Subconsultas escalares y con IN

-- Escalar: un solo valor
SELECT Nombre, Precio FROM dbo.Productos
WHERE Precio > (SELECT AVG(Precio) FROM dbo.Productos);

-- Lista de valores
SELECT Nombre FROM dbo.Productos
WHERE CategoriaId IN (SELECT Id FROM dbo.Categorias WHERE Activa = 1);

EXISTS y subconsultas correlacionadas

-- Productos con al menos una venta
SELECT p.Nombre
FROM dbo.Productos p
WHERE EXISTS (SELECT 1 FROM dbo.Ventas v WHERE v.ProductoId = p.Id);

EXISTS suele ser más eficiente que IN con subconsultas grandes porque deja de buscar al primer match.

CTE: Common Table Expressions

WITH ProductosCaros AS (
  SELECT Id, Nombre, Precio FROM dbo.Productos WHERE Precio > 10000
)
SELECT pc.Nombre, c.Nombre AS Categoria
FROM ProductosCaros pc
LEFT JOIN dbo.Categorias c ON c.Id = pc.Id;

CTE recursiva (jerarquías)

WITH Jerarquia AS (
  SELECT Id, Nombre, JefeId, 1 AS Nivel
  FROM dbo.Empleados WHERE JefeId IS NULL
  UNION ALL
  SELECT e.Id, e.Nombre, e.JefeId, j.Nivel + 1
  FROM dbo.Empleados e
  INNER JOIN Jerarquia j ON e.JefeId = j.Id
)
SELECT REPLICATE('— ', Nivel-1) + Nombre AS Organigrama, Nivel
FROM Jerarquia
OPTION (MAXRECURSION 100);
