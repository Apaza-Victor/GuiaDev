SELECT y alias

SELECT * FROM dbo.Productos;
SELECT Nombre, Precio FROM dbo.Productos;
SELECT Nombre AS Producto, Precio * 1.16 AS PrecioConIva
FROM dbo.Productos;

TOP: el LIMIT de SQL Server

SELECT TOP 10 Nombre, Precio FROM dbo.Productos ORDER BY Precio DESC;

-- Top 10 PORCENTAJE
SELECT TOP 10 PERCENT Nombre FROM dbo.Productos ORDER BY Precio DESC;

-- TOP con empates (incluye empatados en el último lugar)
SELECT TOP 10 WITH TIES Nombre, Precio FROM dbo.Productos
ORDER BY Precio DESC;

TOP sin ORDER BY devuelve filas arbitrarias: casi siempre deben ir juntos.

Filtrar con WHERE

SELECT * FROM dbo.Productos WHERE Precio > 1000;
SELECT * FROM dbo.Productos WHERE Precio BETWEEN 500 AND 1500;
SELECT * FROM dbo.Productos WHERE CategoriaId IN (1, 3, 5);
SELECT * FROM dbo.Productos WHERE Nombre LIKE N'%laptop%';
SELECT * FROM dbo.Productos WHERE CategoriaId IS NULL;

Paginación OFFSET-FETCH

SELECT Nombre, Precio FROM dbo.Productos
ORDER BY Precio DESC
OFFSET 20 ROWS FETCH NEXT 10 ROWS ONLY;   -- página 3 de 10

OFFSET-FETCH exige ORDER BY. Es el equivalente a LIMIT/OFFSET de MySQL.

DISTINCT

SELECT DISTINCT CategoriaId FROM dbo.Productos ORDER BY CategoriaId;
