Transacción básica

BEGIN TRANSACTION;

UPDATE dbo.Cuentas SET Saldo = Saldo - 500 WHERE Id = 1;
UPDATE dbo.Cuentas SET Saldo = Saldo + 500 WHERE Id = 2;

COMMIT TRANSACTION;     -- confirma
-- ROLLBACK TRANSACTION; -- deshace todo si algo falló

Las cuatro propiedades ACID garantizan que ambas actualizaciones ocurran o ninguna: el dinero nunca desaparece.

TRY...CATCH

BEGIN TRY
  BEGIN TRANSACTION;

  UPDATE dbo.Cuentas SET Saldo = Saldo - 500 WHERE Id = 1;
  UPDATE dbo.Cuentas SET Saldo = Saldo + 500 WHERE Id = 2;

  COMMIT TRANSACTION;
END TRY
BEGIN CATCH
  IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
  SELECT
    ERROR_NUMBER()  AS Numero,
    ERROR_SEVERITY() AS Severidad,
    ERROR_STATE()   AS Estado,
    ERROR_LINE()    AS Linea,
    ERROR_MESSAGE() AS Mensaje;
END CATCH;

XACT_ABORT y verificación

SET XACT_ABORT ON;  -- cualquier error deshace TODO automáticamente
BEGIN TRANSACTION;
  -- sentencias...
COMMIT TRANSACTION;

SELECT @@TRANCOUNT AS TransaccionesAbiertas;

SET XACT_ABORT ON al inicio de procedimientos con transacciones evita transacciones zombis ante errores abortantes.
