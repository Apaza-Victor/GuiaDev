DELETE y OUTPUT

DELETE FROM dbo.Clientes WHERE Id = 10;

DELETE TOP (1000) FROM dbo.Logs;   -- borra por lotes

DELETE FROM dbo.Clientes
OUTPUT DELETED.Id, DELETED.Email INTO dbo.ClientesEliminados
WHERE Activo = 0;

MERGE: upsert en una sentencia

MERGE dbo.Inventario AS dest
USING (VALUES (1, 50), (2, 30)) AS src(Id, Stock)
ON dest.Id = src.Id
WHEN MATCHED THEN
  UPDATE SET dest.Stock = src.Stock
WHEN NOT MATCHED THEN
  INSERT (Id, Stock) VALUES (src.Id, src.Stock)
WHEN SOURCE ROW NOT MATCHED BY TARGET AND dest.Stock = 0 THEN
  DELETE;

MERGE combina INSERT/UPDATE/DELETE comparando origen y destino en una sola operación atómica.
