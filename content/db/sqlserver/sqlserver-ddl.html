Crear y eliminar bases de datos

CREATE DATABASE Ventas;
GO

ALTER DATABASE Ventas MODIFY NAME = Ventas2026;
GO

DROP DATABASE IF EXISTS Ventas2026;

Esquemas (schemas)

Un esquema es un contenedor lógico dentro de la base. El predeterminado es dbo. Sirven para organizar y dar permisos por áreas.

CREATE SCHEMA ventas AUTHORIZATION dbo;
GO

CREATE TABLE ventas.Facturas (
  Id INT IDENTITY PRIMARY KEY,
  Total DECIMAL(12,2) NOT NULL
);

CREATE y ALTER TABLE

CREATE TABLE dbo.Categorias (
  Id INT IDENTITY(1,1) PRIMARY KEY,
  Nombre NVARCHAR(50) NOT NULL
);
GO

CREATE TABLE dbo.Productos (
  Id INT IDENTITY(1,1) PRIMARY KEY,
  Nombre NVARCHAR(150) NOT NULL,
  Precio DECIMAL(10,2) NOT NULL CONSTRAINT DF_Productos_Precio DEFAULT(0),
  CategoriaId INT NULL
);
GO

ALTER TABLE dbo.Productos ADD Descripcion NVARCHAR(500) NULL;
ALTER TABLE dbo.Productos ALTER COLUMN Nombre NVARCHAR(200) NOT NULL;
ALTER TABLE dbo.Productos DROP COLUMN Descripcion;

Nombrar los constraints (ej. DF_Productos_Precio) facilita modificarlos después sin depender de nombres autogenerados.

TRUNCATE vs DELETE vs DROP

TRUNCATE TABLE dbo.Productos;   -- vacía rápido, reinicia IDENTITY
DELETE FROM dbo.Productos;      -- fila por fila, permite WHERE
DROP TABLE dbo.Productos;       -- elimina estructura

TRUNCATE no funciona si la tabla tiene llaves foráneas que le apuntan; en ese caso usa DELETE.
