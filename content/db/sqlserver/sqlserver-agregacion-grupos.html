Funciones de agregación

SELECT COUNT(*) AS Total FROM dbo.Productos;
SELECT AVG(Precio) AS Promedio, MIN(Precio) AS Minimo, MAX(Precio) AS Maximo
FROM dbo.Productos;
SELECT SUM(Cantidad) AS Unidades FROM dbo.Ventas;

COUNT(*) cuenta filas; COUNT(Columna) omite nulos; COUNT(DISTINCT Columna) omite repetidos.

GROUP BY y HAVING

SELECT c.Nombre, COUNT(p.Id) AS Productos, ROUND(AVG(p.Precio),2) AS Promedio
FROM dbo.Categorias c
LEFT JOIN dbo.Productos p ON p.CategoriaId = c.Id
GROUP BY c.Nombre
HAVING COUNT(p.Id) >= 3
ORDER BY Productos DESC;

ROLLUP y GROUPING SETS

-- Subtotales por categoría + gran total
SELECT CategoriaId, COUNT(*) AS Total
FROM dbo.Productos
GROUP BY ROLLUP (CategoriaId);

-- Agregaciones personalizadas en una sola pasada
SELECT CategoriaId, COUNT(*) AS Total
FROM dbo.Productos
GROUP BY GROUPING SETS ((CategoriaId), ());

Agregaciones con CASE

SELECT
  COUNT(*) AS Total,
  SUM(CASE WHEN Precio > 1000 THEN 1 ELSE 0 END) AS Caros,
  SUM(CASE WHEN Precio <= 1000 THEN 1 ELSE 0 END) AS Baratos
FROM dbo.Productos;
