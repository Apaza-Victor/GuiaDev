Vistas

CREATE VIEW dbo.vw_ProductosCaros
AS
SELECT Id, Nombre, Precio FROM dbo.Productos WHERE Precio > 10000;
GO

SELECT * FROM dbo.vw_ProductosCaros;

ALTER VIEW dbo.vw_ProductosCaros AS ...;
DROP VIEW dbo.vw_ProductosCaros;

Con CREATE VIEW ... WITH SCHEMABINDING la vista queda amarrada al esquema y puede indexarse (vista indexada/materializada).

Función escalar

CREATE FUNCTION dbo.fn_Iva(@precio DECIMAL(10,2))
RETURNS DECIMAL(10,2)
AS
BEGIN
  RETURN @precio * 1.16;
END;
GO

SELECT Nombre, dbo.fn_Iva(Precio) AS ConIva FROM dbo.Productos;

Función de tabla inline (TVF)

CREATE FUNCTION dbo.fn_ProductosPorCategoria(@categoriaId INT)
RETURNS TABLE
AS
RETURN (
  SELECT Id, Nombre, Precio
  FROM dbo.Productos
  WHERE CategoriaId = @categoriaId
);
GO

SELECT * FROM dbo.fn_ProductosPorCategoria(2);

Las TVF inline se comportan como vistas parametrizadas y el optimizador las trata mejor que las multientencia.
