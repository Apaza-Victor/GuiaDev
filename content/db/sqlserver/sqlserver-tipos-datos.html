Numéricos exactos

CREATE TABLE dbo.EjemploNumerico (
  Edad TINYINT,                    -- 0 a 255
  Contador INT,                    -- entero estándar (~2.147 millones)
  Poblacion BIGINT,                -- enteros gigantes
  Precio DECIMAL(10,2),            -- EXACTO: dinero
  Salario MONEY                    -- alias empresarial de decimal(19,4)
);

DECIMAL(p,s): p = total de dígitos, s = decimales. Para dinero, siempre exacto; nunca FLOAT.

Cadenas: CHAR, VARCHAR y NVARCHAR

CREATE TABLE dbo.EjemploTexto (
  Codigo CHAR(3),                  -- longitud fija ('MXN')
  Nombre VARCHAR(100),             -- variable, no unicode
  Descripcion NVARCHAR(200),       -- variable UNICODE (acentos garantizados)
  Contrato NVARCHAR(MAX)           -- textos enormes
);

N de National: almacena UTF-16 y garantiza acentos y ñ sin problemas de collation. Prefiere NVARCHAR para datos de usuarios.

Fecha y hora

CREATE TABLE dbo.EjemploFechas (
  FechaNacimiento DATE,                       -- solo fecha
  HoraEntrada TIME(7),                        -- solo hora
  RegistradoEn DATETIME2 DEFAULT SYSDATETIME(), -- fecha+hora precisa
  ConZona DATETIMEOFFSET                      -- incluye huso horario
);

Evita DATETIME y SMALLDATETIME en proyectos nuevos: DATETIME2 es más preciso y compatible.

Otros tipos útiles

CREATE TABLE dbo.EjemploOtros (
  Activo BIT,                          -- 1/0 (booleano)
  Guid UNIQUEIDENTIFIER DEFAULT NEWID(), -- identificador global
  Foto VARBINARY(MAX),                 -- binarios/archivos
  XMLDatos XML                         -- documentos XML validados
);

-- IDENTITY: autoincremento estilo SQL Server
CREATE TABLE dbo.Clientes (
  Id INT IDENTITY(1000,5) PRIMARY KEY   -- inicia en 1000, incrementa de 5
);
