PRIMARY KEY + IDENTITY

CREATE TABLE dbo.Clientes (
  Id INT IDENTITY(1,1) CONSTRAINT PK_Clientes PRIMARY KEY,
  Email NVARCHAR(120) NOT NULL
);

FOREIGN KEY y acciones referenciales

CREATE TABLE dbo.Pedidos (
  Id INT IDENTITY(1,1) PRIMARY KEY,
  ClienteId INT NOT NULL,
  Fecha DATE NOT NULL CONSTRAINT DF_Pedidos_Fecha DEFAULT(GETDATE()),
  CONSTRAINT FK_Pedidos_Clientes FOREIGN KEY (ClienteId)
    REFERENCES dbo.Clientes(Id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

Acciones disponibles: CASCADE, SET NULL, SET DEFAULT y NO ACTION (bloquea).

UNIQUE, CHECK y NOT NULL

CREATE TABLE dbo.Empleados (
  Id INT IDENTITY PRIMARY KEY,
  Curp CHAR(18) NOT NULL CONSTRAINT UQ_Empleados_Curp UNIQUE,
  Nombre NVARCHAR(80) NOT NULL,
  Salario DECIMAL(10,2) CONSTRAINT CK_Empleados_Salario CHECK (Salario > 0),
  Edad TINYINT CONSTRAINT CK_Empleados_Edad CHECK (Edad BETWEEN 18 AND 100)
);

Administrar restricciones

-- Ver restricciones de una tabla
SELECT name, type_desc FROM sys.objects
WHERE parent_object_id = OBJECT_ID('dbo.Empleados');

-- Eliminar y volver a crear
ALTER TABLE dbo.Empleados DROP CONSTRAINT CK_Empleados_Edad;
ALTER TABLE dbo.Empleados ADD CONSTRAINT CK_Empleados_Edad CHECK (Edad >= 16);
