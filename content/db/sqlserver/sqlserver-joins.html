INNER JOIN

SELECT p.Nombre, c.Nombre AS Categoria, p.Precio
FROM dbo.Productos p
INNER JOIN dbo.Categorias c ON p.CategoriaId = c.Id;

LEFT / RIGHT / FULL OUTER

-- Todos los productos aunque no tengan venta
SELECT p.Nombre, v.Fecha, v.Cantidad
FROM dbo.Productos p
LEFT JOIN dbo.Ventas v ON v.ProductoId = p.Id;

-- Productos jamás vendidos
SELECT p.Nombre
FROM dbo.Productos p
LEFT JOIN dbo.Ventas v ON v.ProductoId = p.Id
WHERE v.Id IS NULL;

-- FULL OUTER: todo de ambos lados
SELECT p.Nombre, v.Fecha
FROM dbo.Productos p
FULL OUTER JOIN dbo.Ventas v ON v.ProductoId = p.Id;

En un LEFT JOIN, filtros sobre la tabla derecha van mejor en el ON; ponerlos en el WHERE puede convertirlo en INNER.

CROSS JOIN y auto-join

SELECT e.Nombre, t.Horario
FROM dbo.Empleados e CROSS JOIN dbo.Turnos t;

-- Empleado y su jefe (misma tabla)
SELECT e.Nombre AS Empleado, j.Nombre AS Jefe
FROM dbo.Empleados e
LEFT JOIN dbo.Empleados j ON e.JefeId = j.Id;

Varias tablas encadenadas

SELECT c.Nombre AS Cliente, p.Id AS Pedido, pr.Nombre AS Producto
FROM dbo.Pedidos p
INNER JOIN dbo.Clientes c ON p.ClienteId = c.Id
INNER JOIN dbo.DetallesPedido d ON d.PedidoId = p.Id
INNER JOIN dbo.Productos pr ON d.ProductoId = pr.Id
ORDER BY c.Nombre;
