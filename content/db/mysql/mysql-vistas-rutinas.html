Vistas (VIEW)

Una vista es una consulta guardada que se usa como si fuera una tabla.

CREATE VIEW vw_productos_caros AS
SELECT nombre, precio FROM productos WHERE precio > 10000;

SELECT * FROM vw_productos_caros;

DROP VIEW IF EXISTS vw_productos_caros;

Procedimientos almacenados

DELIMITER //

CREATE PROCEDURE sp_subir_precios(IN p_categoria INT, IN p_porcentaje DECIMAL(5,2))
BEGIN
  UPDATE productos
  SET precio = precio * (1 + p_porcentaje / 100)
  WHERE categoria_id = p_categoria;
END //

DELIMITER ;

CALL sp_subir_precios(1, 10.00);   -- sube 10% a la categoría 1

DELIMITER // cambia el separador temporalmente para que MySQL no interprete los ; internos como fin del procedimiento.

Funciones almacenadas

DELIMITER //
CREATE FUNCTION fn_iva(precio DECIMAL(10,2)) RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
  RETURN precio * 1.16;
END //
DELIMITER ;

SELECT nombre, fn_iva(precio) AS con_iva FROM productos;

Triggers

DELIMITER //
CREATE TRIGGER trg_descuenta_stock
AFTER INSERT ON ventas
FOR EACH ROW
BEGIN
  UPDATE productos
  SET stock = stock - NEW.cantidad
  WHERE id = NEW.producto_id;
END //
DELIMITER ;

Los triggers se ejecutan en cada fila afectada y pueden degradar el rendimiento en cargas masivas. Úsalos solo cuando la regla sea imprescindible.
