PRIMARY KEY y AUTO_INCREMENT

La llave primaria identifica cada fila de forma única. Con AUTO_INCREMENT, MySQL genera el valor por ti.

CREATE TABLE clientes (
  id INT AUTO_INCREMENT PRIMARY KEY,
  email VARCHAR(120) NOT NULL UNIQUE
);

FOREIGN KEY

La llave foránea vincula una columna con la llave primaria de otra tabla, impidiendo valores huérfanos.

CREATE TABLE pedidos (
  id INT AUTO_INCREMENT PRIMARY KEY,
  cliente_id INT NOT NULL,
  FOREIGN KEY (cliente_id) REFERENCES clientes(id)
    ON DELETE CASCADE     -- al borrar el cliente, se borran sus pedidos
    ON UPDATE CASCADE     -- al cambiar su id, se propaga
);

Opciones de eliminación: CASCADE (propaga), SET NULL (deja huérfano permitido), RESTRICT/NO ACTION (bloquea el borrado).

UNIQUE, NOT NULL, DEFAULT y CHECK

CREATE TABLE empleados (
  id INT AUTO_INCREMENT PRIMARY KEY,
  curp CHAR(18) NOT NULL UNIQUE,          -- no repetido
  nombre VARCHAR(80) NOT NULL,            -- obligatorio
  salario DECIMAL(10,2) DEFAULT 15000.00, -- valor por defecto
  edad TINYINT CHECK (edad >= 18),        -- regla de validación (MySQL 8.0.16+)
  departamento VARCHAR(40) DEFAULT 'General'
);

Restricciones con nombre

Nombrar las restricciones facilita eliminarlas o depurarlas después.

CREATE TABLE pagos (
  id INT AUTO_INCREMENT PRIMARY KEY,
  empleado_id INT NOT NULL,
  monto DECIMAL(10,2) NOT NULL,
  CONSTRAINT fk_pagos_empleado FOREIGN KEY (empleado_id)
    REFERENCES empleados(id),
  CONSTRAINT ck_pagos_monto CHECK (monto > 0)
);

ALTER TABLE pagos DROP FOREIGN KEY fk_pagos_empleado;
