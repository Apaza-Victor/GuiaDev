Funciones de agregación

SELECT COUNT(*) AS total_productos FROM productos;
SELECT AVG(precio) AS precio_promedio FROM productos;
SELECT MIN(precio) AS mas_barato, MAX(precio) AS mas_caro FROM productos;
SELECT SUM(stock) AS unidades_totales FROM productos;

COUNT(*) cuenta filas (incluye nulos); COUNT(columna) ignora los NULL de esa columna.

GROUP BY

SELECT categoria_id, COUNT(*) AS cantidad, ROUND(AVG(precio),2) AS promedio
FROM productos
GROUP BY categoria_id;

HAVING vs WHERE

SELECT categoria_id, COUNT(*) AS total
FROM productos
WHERE precio > 100          -- filtra FILAS antes de agrupar
GROUP BY categoria_id
HAVING COUNT(*) >= 3        -- filtra GRUPOS después de agrupar
ORDER BY total DESC;

WHERE no puede usar funciones de agregación; para eso existe HAVING.

GROUP_CONCAT y WITH ROLLUP

-- Unir valores de un grupo en una sola cadena
SELECT categoria_id, GROUP_CONCAT(nombre SEPARATOR ', ') AS productos
FROM productos GROUP BY categoria_id;

-- Subtotales automáticos por grupo
SELECT categoria_id, COUNT(*) AS total
FROM productos
GROUP BY categoria_id WITH ROLLUP;
