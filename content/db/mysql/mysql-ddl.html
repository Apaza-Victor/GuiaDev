Crear base de datos

CREATE DATABASE tienda CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
SHOW DATABASES;
USE tienda;

utf8mb4 soporta emojis y todos los caracteres Unicode. Es el estándar recomendado hoy en día.

CREATE TABLE

CREATE TABLE categorias (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL
);

CREATE TABLE productos (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  precio DECIMAL(10,2) NOT NULL DEFAULT 0.00,
  stock INT UNSIGNED DEFAULT 0,
  categoria_id INT,
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (categoria_id) REFERENCES categorias(id)
);

SHOW TABLES;
DESCRIBE productos;   -- estructura de la tabla

ALTER TABLE

-- Agregar columna
ALTER TABLE productos ADD COLUMN descripcion TEXT AFTER nombre;

-- Modificar tipo
ALTER TABLE productos MODIFY COLUMN nombre VARCHAR(150) NOT NULL;

-- Renombrar columna (MySQL 8+)
ALTER TABLE productos RENAME COLUMN descripcion TO detalle;

-- Eliminar columna
ALTER TABLE productos DROP COLUMN detalle;

-- Renombrar tabla
RENAME TABLE productos TO articulos;

Vaciar y eliminar

TRUNCATE TABLE articulos;   -- borra TODAS las filas rápido, reinicia el AUTO_INCREMENT
DROP TABLE articulos;       -- elimina también la estructura
DROP DATABASE IF EXISTS prueba;

DROP es irreversible. TRUNCATE no dispara triggers ni respeta WHERE: si necesitas condiciones, usa DELETE.
