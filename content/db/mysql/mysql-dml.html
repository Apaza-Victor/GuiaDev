INSERT

-- Una fila
INSERT INTO clientes (email) VALUES ('ana@mail.com');

-- Varias filas de una vez (más eficiente)
INSERT INTO clientes (email) VALUES
  ('luis@mail.com'),
  ('maria@mail.com');

-- Especificando id manualmente
INSERT INTO clientes (id, email) VALUES (10, 'leo@mail.com');

Si omites una columna con DEFAULT o nullable, tomará su valor por defecto.

UPDATE

UPDATE clientes SET email = 'ana-nueva@mail.com' WHERE id = 1;

-- Actualización masiva con cálculo
UPDATE productos SET precio = precio * 1.10 WHERE categoria_id = 2;

-- UPDATE con JOIN
UPDATE productos p
INNER JOIN categorias c ON p.categoria_id = c.id
SET p.stock = 0
WHERE c.nombre = 'Ofertas';

Un UPDATE sin WHERE modifica TODAS las filas. Ejecuta antes el SELECT equivalente para revisar qué filas afectará.

DELETE

DELETE FROM clientes WHERE id = 10;

-- Eliminar con condición relacional
DELETE p FROM productos p
LEFT JOIN categorias c ON p.categoria_id = c.id
WHERE c.id IS NULL;   -- productos sin categoría válida

DELETE registra en el log cada fila borrada (es recuperable dentro de una transacción); TRUNCATE vacía la tabla completa sin ese registro detallado.

INSERT INTO clientes_archivo (id, email)
SELECT id, email FROM clientes WHERE activo = FALSE;

Ideal para respaldos parciales o migraciones entre tablas con la misma estructura.
