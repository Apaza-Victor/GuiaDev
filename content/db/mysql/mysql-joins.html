INNER JOIN

Devuelve solo las filas que tienen coincidencia en ambas tablas.

SELECT p.nombre, c.nombre AS categoria, p.precio
FROM productos p
INNER JOIN categorias c ON p.categoria_id = c.id;

LEFT y RIGHT JOIN

-- Todos los productos, con o sin venta asociada
SELECT p.nombre, v.fecha, v.cantidad
FROM productos p
LEFT JOIN ventas v ON v.producto_id = p.id;

-- Productos que NUNCA se han vendido
SELECT p.nombre
FROM productos p
LEFT JOIN ventas v ON v.producto_id = p.id
WHERE v.id IS NULL;

El patrón LEFT JOIN ... WHERE t2.id IS NULL es la forma clásica de encontrar registros sin relación.

CROSS JOIN y auto-join

-- Producto cartesiano: cada empleado con cada turno
SELECT e.nombre, t.horario
FROM empleados e CROSS JOIN turnos t;

-- Auto-join: empleado y su jefe
SELECT e.nombre AS empleado, j.nombre AS jefe
FROM empleados e
LEFT JOIN empleados j ON e.jefe_id = j.id;

Más de dos tablas

SELECT c.nombre AS cliente, p.id AS pedido, pr.nombre AS producto
FROM pedidos p
INNER JOIN clientes c ON p.cliente_id = c.id
INNER JOIN detalles_pedido d ON d.pedido_id = p.id
INNER JOIN productos pr ON d.producto_id = pr.id
ORDER BY c.nombre;
