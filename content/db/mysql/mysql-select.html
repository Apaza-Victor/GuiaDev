SELECT básico y alias

SELECT * FROM productos;                       -- todas las columnas
SELECT nombre, precio FROM productos;          -- columnas específicas
SELECT nombre AS producto, precio * 1.16 AS precio_con_iva FROM productos;

Filtrar con WHERE

SELECT * FROM productos WHERE precio > 1000;
SELECT * FROM productos WHERE precio BETWEEN 500 AND 1500;
SELECT * FROM productos WHERE categoria_id IN (1, 3, 5);
SELECT * FROM productos WHERE nombre LIKE '%laptop%';   -- contiene
SELECT * FROM productos WHERE nombre LIKE 'L%';         -- empieza con L
SELECT * FROM productos WHERE categoria_id IS NULL;     -- sin categoría

LIKE: % reemplaza cualquier cadena, _ un solo carácter. Para buscar NULL nunca uses = NULL, usa IS NULL.

Ordenar y paginar

SELECT nombre, precio FROM productos
ORDER BY precio DESC, nombre ASC
LIMIT 10 OFFSET 20;   -- página 3, de 10 en 10

LIMIT n OFFSET m es la base de toda paginación: OFFSET = (página − 1) × tamaño_de_página.

DISTINCT y combinaciones

SELECT DISTINCT categoria_id FROM productos;

-- Condiciones compuestas: AND / OR / NOT
SELECT * FROM productos
WHERE (precio > 1000 AND stock > 0)
   OR categoria_id IS NULL;
