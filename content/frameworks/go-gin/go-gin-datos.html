6.1 Path parameters

Definición

Los path parameters son segmentos de la URL que capturan valores dinámicos. Se definen con dos puntos (:parametro) y se acceden con c.Param().

r.GET("/usuarios/:id", func(c *gin.Context) {
    id := c.Param("id")
    c.JSON(200, gin.H{"id": id})
})

// Múltiples parámetros
r.GET("/usuarios/:id/posts/:postID/comentarios/:comentarioID", func(c *gin.Context) {
    c.JSON(200, gin.H{
        "usuario":     c.Param("id"),
        "post":        c.Param("postID"),
        "comentario":   c.Param("comentarioID"),
    })
})

// Wildcard: captura cualquier ruta restante
r.GET("/archivos/*ruta", func(c *gin.Context) {
    c.JSON(200, gin.H{"ruta_completa": c.Param("ruta")})
})

Conversión de tipos

r.GET("/usuarios/:id", func(c *gin.Context) {
    idStr := c.Param("id")
    id, err := strconv.Atoi(idStr) // string -> int
    if err != nil {
        c.JSON(400, gin.H{"error": "id debe ser numérico"})
        return
    }
    c.JSON(200, gin.H{"id_entero": id, "id_aumentado": id + 1})
})

Conteo de parámetros

SintaxisResultado
/:idCaptura un segmento
/*rutaCaptura todo lo restante
/:id/:pageCaptura varios segmentos

6.2 Query strings

Definición

Los query strings son pares clave-valor que van después de ? en la URL. Se usan para filtros, paginación y opciones.

// URL: /buscar?q=golang&page=2&limite=10
r.GET("/buscar", func(c *gin.Context) {
    q := c.Query("q")             // "golang"
    page := c.DefaultQuery("page", "1")  // "2" o "1" por defecto
    limite := c.Query("limite")   // "10"

    c.JSON(200, gin.H{
        "q":      q,
        "page":   page,
        "limite": limite,
    })
})

Búsqueda por URL

r.GET("/productos", func(c *gin.Context) {
    categoria := c.Query("categoria")      // opcional
    precioMax := c.DefaultQuery("precio_max", "999999")
    orden := c.DefaultQuery("orden", "desc")

    c.JSON(200, gin.H{
        "filtro_categoria": categoria,
        "precio_maximo":    precioMax,
        "orden":            orden,
    })
})
// GET /productos?categoria=ropa -> filtra solo ropa
// GET /productos -> devuelve todo con defaults

6.3 Formularios

Recibir formularios con c.PostForm()

r.POST("/registro", func(c *gin.Context) {
    nombre := c.PostForm("nombre")
    email := c.PostForm("email")
    edad := c.PostForm("edad")

    // PostForm con valor por defecto
    genero := c.DefaultPostForm("genero", "no_especificado")

    c.JSON(200, gin.H{
        "nombre": nombre,
        "email":  email,
        "edad":   edad,
        "genero": genero,
    })
})

Form + Query combinados

r.POST("/alternativa", func(c *gin.Context) {
    // Otros métodos para obtener datos
    v1 := c.Request.FormValue("campo1")        // forma del stdlib
    v2 := c.GetPostForm("campo2")              // (valor, existe)

    valor, existe := c.GetPostForm("campo2")
    if !existe {
        c.JSON(400, gin.H{"error": "campo2 es requerido"})
        return
    }
    c.JSON(200, gin.H{"campo1": v1, "campo2": valor})
})

6.4 Validación de entrada

Validar con formatos

Combina ShouldBind con tags para validar formularios y queries:

type BusquedaLibro struct {
    Titulo  string `form:"titulo" binding:"omitempty"`
    Autor   string `form:"autor" binding:"omitempty"`
    Anio    int    `form:"anio" binding:"gte=1000,lte=2100,omitempty"`
    Pagina  int    `form:"page" binding:"gte=1"`
}

r.GET("/libros", func(c *gin.Context) {
    var busqueda BusquedaLibro
    if err := c.ShouldBindQuery(&busqueda); err != nil {
        c.JSON(400, gin.H{"error": "filtros inválidos", "detalle": err.Error()})
        return
    }
    c.JSON(200, busqueda)
})

Validación manual

func esEmailValido(email string) bool {
    return strings.Contains(email, "@") && strings.Contains(email, ".")
}

r.POST("/contacto", func(c *gin.Context) {
    email := c.PostForm("email")
    mensaje := c.PostForm("mensaje")

    if len(mensaje) < 10 {
        c.JSON(400, gin.H{"error": "el mensaje debe tener al menos 10 caracteres"})
        return
    }
    if !esEmailValido(email) {
        c.JSON(400, gin.H{"error": "email no válido"})
        return
    }
    c.JSON(200, gin.H{"recibido": true})
})

Buenas prácticas

  • Valida siempre los datos de entrada, nunca confíes en el cliente
  • Devuelve errores descriptivos en español para el usuario final
  • Usa binding: para reglas comunes y validación manual para lógica compleja
  • Convierte strings a tipos con strconv manejando el error

6.5 Ejemplo práctico

API de productos con filtros

package main

import (
    "fmt"
    "strconv"
    "net/http"
    "github.com/gin-gonic/gin"
)

type Producto struct {
    ID       int     `json:"id"`
    Nombre   string  `json:"nombre"`
    Categoria string `json:"categoria"`
    Precio   float64 `json:"precio"`
}

var catalogo = []Producto{
    {ID: 1, Nombre: "Laptop", Categoria: "tecnologia", Precio: 999.99},
    {ID: 2, Nombre: "Mouse", Categoria: "tecnologia", Precio: 29.99},
    {ID: 3, Nombre: "Camisa", Categoria: "ropa", Precio: 19.00},
}

func main() {
    r := gin.Default()

    // LISTAR con filtros y paginación
    r.GET("/productos", func(c *gin.Context) {
        categoria := c.Query("categoria")
        precioMax, _ := strconv.ParseFloat(c.DefaultQuery("precio_max", "999999"), 64)
        pagina, _ := strconv.Atoi(c.DefaultQuery("page", "1"))
        porPagina := 10

        var resultados []Producto
        for _, p := range catalogo {
            if categoria != "" && p.Categoria != categoria {
                continue
            }
            if p.Precio > precioMax {
                continue
            }
            resultados = append(resultados, p)
        }

        // Paginación simple
        inicio := (pagina - 1) * porPagina
        if inicio > len(resultados) {
            inicio = len(resultados)
        }
        fin := inicio + porPagina
        if fin > len(resultados) {
            fin = len(resultados)
        }

        c.JSON(http.StatusOK, gin.H{
            "total":     len(resultados),
            "pagina":    pagina,
            "productos": resultados[inicio:fin],
        })
    })

    // OBTENER por ID (path parameter)
    r.GET("/productos/:id", func(c *gin.Context) {
        idStr := c.Param("id")
        id, err := strconv.Atoi(idStr)
        if err != nil || id <= 0 {
            c.JSON(http.StatusBadRequest, gin.H{"error": "id inválido"})
            return
        }
        for _, p := range catalogo {
            if p.ID == id {
                c.JSON(http.StatusOK, p)
                return
            }
        }
        c.JSON(http.StatusNotFound, gin.H{"error": "producto no encontrado"})
    })

    r.Run(":8080")
}

Resumen

**Lo que aprendiste:** - Los path parameters (:id) se obtienen con c.Param() - Las query strings se obtienen con c.Query() y c.DefaultQuery() - Los formularios se reciben con c.PostForm() - Usa strconv para convertir strings a números validos - ShouldBindQuery valida filtros con tags binding - Valida siempre las entradas y responde errores descriptivos
