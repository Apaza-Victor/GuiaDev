9.1 Recibir formularios

c.PostForm()

Los formularios enviados con Content-Type: application/x-www-form-urlencoded se leen con c.PostForm():

r.POST("/contacto", func(c *gin.Context) {
    nombre := c.PostForm("nombre")
    email := c.PostForm("email")
    mensaje := c.DefaultPostForm("mensaje", "sin mensaje")

    c.JSON(200, gin.H{
        "nombre":  nombre,
        "email":   email,
        "mensaje": mensaje,
    })
})

Leer todos los campos

r.POST("/multi", func(c *gin.Context) {
    // GetPostForm devuelve también si el campo existe
    valor, existe := c.GetPostForm("campo_requerido")
    if !existe {
        c.JSON(400, gin.H{"error": "campo_requerido es obligatorio"})
        return
    }

    // PostFormArray para múltiples valores con el mismo nombre
    intereses := c.PostFormArray("intereses")

    // PostFormMap para datos con prefijo como "telefono[casa]"
    telefonos := c.PostFormMap("telefono")

    c.JSON(200, gin.H{
        "campo":     valor,
        "intereses": intereses,
        "telefonos": telefonos,
    })
})

Binding de formulario a struct

type Contacto struct {
    Nombre string `form:"nombre" binding:"required"`
    Email  string `form:"email" binding:"required,email"`
    Edad   int    `form:"edad" binding:"gte=18"`
}

r.POST("/registro-form", func(c *gin.Context) {
    var contacto Contacto
    if err := c.ShouldBind(&contacto); err != nil {
        c.JSON(400, gin.H{"error": err.Error()})
        return
    }
    c.JSON(201, contacto)
})

9.2 Subida de archivos

c.FormFile()

Para subir archivos, el formulario debe usar enctype="multipart/form-data". Gin expone el archivo con c.FormFile():

r.POST("/subir", func(c *gin.Context) {
    // Archivo con nombre de campo "archivo"
    archivo, err := c.FormFile("archivo")
    if err != nil {
        c.JSON(400, gin.H{"error": "archivo requerido"})
        return
    }

    // Guardar en el servidor
    ruta := "./uploads/" + archivo.Filename
    if err := c.SaveUploadedFile(archivo, ruta); err != nil {
        c.JSON(500, gin.H{"error": "no se pudo guardar"})
        return
    }

    c.JSON(200, gin.H{
        "mensaje":  "archivo subido",
        "nombre":   archivo.Filename,
        "tamaño":   archivo.Size, // en bytes
        "cabecera": archivo.Header.Get("Content-Type"),
    })
})

Restricciones de tamaño

// Límite global (máximo 8 MB)
r.MaxMultipartMemory = 8 << 20 // 8 MB

// O por handler con http.MaxBytesReader
r.POST("/subir", func(c *gin.Context) {
    c.Request.Body = http.MaxBytesReader(
        c.Writer, c.Request.Body, 5<<20, // 5 MB
    )
    // ...
})

Validar tipo y extensión

func archivoImagenValido(c *gin.Context) bool {
    archivo, err := c.FormFile("imagen")
    if err != nil {
        return false
    }
    tipo := archivo.Header.Get("Content-Type")
    permitidos := map[string]bool{
        "image/jpeg": true,
        "image/png":  true,
        "image/webp": true,
    }
    return permitidos[tipo]
}
Nunca guardes archivos con el nombre original del usuario sin sanitizarlo. Usa nombres generados (UUID) para evitar path traversal y sobreescrituras maliciosas.

9.3 Archivos estáticos

Servir el directorio completo

r := gin.Default()

// Sirve todo el directorio ./public en /public/*
r.Static("/public", "./public")

// Sirve un archivo individual
r.StaticFile("/favicon.ico", "./assets/favicon.ico")

// Sirve templates HTML (assets en memoria con embed)
r.StaticFS("/assets", http.Dir("./assets"))

Estructura de carpetas

public/
├── img/
│   ├── logo.png
│   └── avatar.png
├── css/
│   └── estilos.css
├── js/
│   └── app.js
└── index.html
// Uso: el navegador accede a
// http://localhost:8080/public/img/logo.png
// http://localhost:8080/public/css/estilos.css
// http://localhost:8080/public/js/app.js

Middleware para servir con caché

func cachéEstaticos() gin.HandlerFunc {
    return func(c *gin.Context) {
        c.Header("Cache-Control", "public, max-age=86400")
        c.Next()
    }
}

r.Use(cachéEstaticos())
r.Static("/public", "./public")

9.4 Multipart form

Campos y archivos juntos

Un formulario multipart puede enviar campos simples y varios archivos en una sola petición:

r.POST("/publicar", func(c *gin.Context) {
    titulo := c.PostForm("titulo")
    descripcion := c.DefaultPostForm("descripcion", "")

    // Un archivo
    principal, _ := c.FormFile("imagen_principal")

    // Varios archivos con el mismo nombre de campo
    galeria, err := c.MultipartForm()
    if err != nil {
        c.JSON(400, gin.H{"error": "formato inválido"})
        return
    }
    fotos := galeria.File["fotos"]

    c.JSON(200, gin.H{
        "titulo":          titulo,
        "descripcion":     descripcion,
        "imagen_principal": principal.Filename,
        "cantidad_fotos":  len(fotos),
    })
})

Guardar todos los archivos

r.POST("/galeria", func(c *gin.Context) {
    form, err := c.MultipartForm()
    if err != nil {
        c.JSON(400, gin.H{"error": err.Error()})
        return
    }
    archivos := form.File["imagenes"]

    guardados := []string{}
    for _, archivo := range archivos {
        ruta := "./uploads/" + archivo.Filename
        if err := c.SaveUploadedFile(archivo, ruta); err == nil {
            guardados = append(guardados, ruta)
        }
    }
    c.JSON(200, gin.H{
        "total_subidos": len(guardados),
        "archivos":      guardados,
    })
})

Probar con curl

curl -X POST http://localhost:8080/galeria \
  -F "imagenes=@foto1.png" \
  -F "imagenes=@foto2.jpg" \
  -F "titulo=Vacaciones"

9.5 Ejemplo práctico

API completa: formulario + archivos + estáticos

package main

import (
    "net/http"
    "os"
    "github.com/gin-gonic/gin"
)

func main() {
    // Crear carpeta de uploads
    os.MkdirAll("./uploads", 0755)

    r := gin.Default()
    r.MaxMultipartMemory = 8 << 20 // 8 MB

    // Servir archivos estáticos
    r.Static("/uploads", "./uploads")

    // Crear un avatar (campo + archivo)
    r.POST("/perfil/:id/avatar", func(c *gin.Context) {
        id := c.Param("id")

        // Validar que venga archivo
        archivo, err := c.FormFile("avatar")
        if err != nil {
            c.JSON(http.StatusBadRequest, gin.H{"error": "sube un avatar"})
            return
        }

        // Validar extensión (PNG/JPG)
        tipo := archivo.Header.Get("Content-Type")
        if tipo != "image/png" && tipo != "image/jpeg" {
            c.JSON(http.StatusBadRequest, gin.H{
                "error":   "solo se permiten PNG o JPG",
                "recibido": tipo,
            })
            return
        }

        // Guardar con nombre seguro
        nombreUnico := "avatar_" + id + "_" + archivo.Filename
        c.SaveUploadedFile(archivo, "./uploads/"+nombreUnico)

        c.JSON(http.StatusOK, gin.H{
            "mensaje":      "avatar actualizado",
            "url":          "/uploads/" + nombreUnico,
            "tamaño_bytes": archivo.Size,
        })
    })

    // Formulario de publicacion completa
    r.POST("/posts", func(c *gin.Context) {
        titulo := c.PostForm("titulo")
        contenido := c.PostForm("contenido")
        etiqueta := c.DefaultPostForm("etiqueta", "general")

        // Archivos opcionales
        var adjuntos []string
        form, _ := c.MultipartForm()
        if form != nil {
            for _, f := range form.File["adjuntos"] {
                c.SaveUploadedFile(f, "./uploads/"+f.Filename)
                adjuntos = append(adjuntos, f.Filename)
            }
        }

        c.JSON(http.StatusCreated, gin.H{
            "titulo":     titulo,
            "contenido":  contenido,
            "etiqueta":   etiqueta,
            "adjuntos":   adjuntos,
        })
    })

    r.Run(":8080")
}

Probar los endpoints

# Subir avatar
curl -X POST http://localhost:8080/perfil/1/avatar \
  -F "avatar=@mi_foto.png"

# Crear post con adjuntos
curl -X POST http://localhost:8080/posts \
  -F "titulo=Mi primer post" \
  -F "contenido=Hola mundo" \
  -F "etiqueta=go" \
  -F "adjuntos=@doc.pdf"

# Ver archivo subido
curl http://localhost:8080/uploads/mi_foto.png
Siempre limita el tamaño de las subidas y valida el tipo de archivo antes de guardarlo.

Resumen

**Lo que aprendiste:** - c.PostForm() lee campos de formularios x-www-form-urlencoded - c.FormFile() obtiene un archivo de un multipart - c.SaveUploadedFile() guarda archivos en el servidor - c.MultipartForm() accede a todos los campos y archivos - r.Static() sirve directorios de archivos estáticos - Valida tipo, tamaño y evade nombres seguros de archivo
