7.1 CRUD en memoria

Definición

El almacenamiento en memoria usa estructuras de Go (slices, maps) para guardar datos. Es ideal para prototipos, aprendizaje y APIs que no requieren persistencia real.

package main

import (
    "strconv"
    "net/http"
    "github.com/gin-gonic/gin"
)

type Usuario struct {
    ID     int    `json:"id"`
    Nombre string `json:"nombre"`
    Email  string `json:"email"`
}

// Almacenamiento en memoria
var usuarios = map[int]Usuario{}
var siguienteID = 1

func main() {
    r := gin.Default()

    // CREATE
    r.POST("/usuarios", func(c *gin.Context) {
        var u Usuario
        if err := c.ShouldBindJSON(&u); err != nil {
            c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
            return
        }
        u.ID = siguienteID
        usuarios[u.ID] = u
        siguienteID++
        c.JSON(http.StatusCreated, u)
    })

    // READ (todos)
    r.GET("/usuarios", func(c *gin.Context) {
        lista := []Usuario{}
        for _, u := range usuarios {
            lista = append(lista, u)
        }
        c.JSON(http.StatusOK, lista)
    })

    // READ (uno)
    r.GET("/usuarios/:id", func(c *gin.Context) {
        id, _ := strconv.Atoi(c.Param("id"))
        u, existe := usuarios[id]
        if !existe {
            c.JSON(http.StatusNotFound, gin.H{"error": "no existe"})
            return
        }
        c.JSON(http.StatusOK, u)
    })

    // UPDATE
    r.PUT("/usuarios/:id", func(c *gin.Context) {
        id, _ := strconv.Atoi(c.Param("id"))
        var u Usuario
        if err := c.ShouldBindJSON(&u); err != nil {
            c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
            return
        }
        u.ID = id
        usuarios[id] = u
        c.JSON(http.StatusOK, u)
    })

    // DELETE
    r.DELETE("/usuarios/:id", func(c *gin.Context) {
        id, _ := strconv.Atoi(c.Param("id"))
        delete(usuarios, id)
        c.Status(http.StatusNoContent)
    })

    r.Run(":8080")
}

7.2 Conexión con database/sql

database/sql

La standard library database/sql proporciona una interfaz unificada para bases de datos SQL. Necesitas un driver específico (mysql, postgres, sqlite).

# Instalar driver de PostgreSQL
go get github.com/lib/pq

# Instalar driver de MySQL
go get github.com/go-sql-driver/mysql

# Instalar driver de SQLite
go get github.com/mattn/go-sqlite3

Conexión básica

package db

import (
    "database/sql"
    _ "github.com/lib/pq" // driver de PostgreSQL
)

func Conectar() (*sql.DB, error) {
    dsn := "postgres://usuario:password@localhost:5432/miapi?sslmode=disable"
    db, err := sql.Open("postgres", dsn)
    if err != nil {
        return nil, err
    }
    if err := db.Ping(); err != nil {
        return nil, err
    }
    return db, nil
}

Operaciones básicas

// CREATE
res, err := db.Exec(
    "INSERT INTO usuarios (nombre, email) VALUES ($1, $2)",
    "Ana", "ana@mail.com",
)

// READ
var nombre, email string
err = db.QueryRow(
    "SELECT nombre, email FROM usuarios WHERE id = $1", 1,
).Scan(&nombre, &email)

// READ (varios)
rows, err := db.Query("SELECT id, nombre FROM usuarios")
for rows.Next() {
    var id int
    var nombre string
    rows.Scan(&id, &nombre)
}
rows.Close()

// UPDATE
db.Exec("UPDATE usuarios SET nombre = $1 WHERE id = $2", "Ana María", 1)

// DELETE
db.Exec("DELETE FROM usuarios WHERE id = $1", 1)

7.3 ORM con GORM

Qué es un ORM

Un ORM (Object-Relational Mapping) mapea tablas a structs de Go, eliminando el SQL manual. GORM es el ORM más popular para Go.

go get -u gorm.io/gorm
go get -u gorm.io/driver/sqlite
go get -u gorm.io/driver/postgres

Modelo con GORM

package models

import "gorm.io/gorm"

type Usuario struct {
    gorm.Model          // incluye ID, CreatedAt, UpdatedAt, DeletedAt
    Nombre  string      `gorm:"size:100;not null" json:"nombre"`
    Email   string      `gorm:"uniqueIndex;not null" json:"email"`
    Edad    int         `json:"edad"`
    Posts   []Post      `json:"posts,omitempty"`
}

type Post struct {
    gorm.Model
    Titulo string  `json:"titulo"`
    UsuarioID uint  `json:"usuario_id"`
}

Conexión y migración

package main

import (
    "gorm.io/driver/sqlite"
    "gorm.io/gorm"
    "miapi/models"
)

func main() {
    db, err := gorm.Open(sqlite.Open("miapi.db"), &gorm.Config{})
    if err != nil {
        panic("fallo la conexión: " + err.Error())
    }

    // AutoMigrate crea o actualiza las tablas
    db.AutoMigrate(&models.Usuario{}, &models.Post{})
}

7.4 Operaciones CRUD en SQL

CRUD con database/sql en handlers

package main

import (
    "database/sql"
    "net/http"
    "strconv"
    "github.com/gin-gonic/gin"
)

func main() {
    db, _ := db.Conectar()
    defer db.Close()

    r := gin.Default()

    // CREATE
    r.POST("/usuarios", func(c *gin.Context) {
        var u struct {
            Nombre string `json:"nombre"`
            Email  string `json:"email"`
        }
        if err := c.ShouldBindJSON(&u); err != nil {
            c.JSON(400, gin.H{"error": err.Error()})
            return
        }
        res, err := db.Exec(
            "INSERT INTO usuarios (nombre, email) VALUES ($1, $2)",
            u.Nombre, u.Email,
        )
        if err != nil {
            c.JSON(500, gin.H{"error": "no se pudo crear"})
            return
        }
        id, _ := res.LastInsertId()
        c.JSON(http.StatusCreated, gin.H{"id": id})
    })

    // READ (uno)
    r.GET("/usuarios/:id", func(c *gin.Context) {
        id, _ := strconv.Atoi(c.Param("id"))
        var usuario struct {
            ID     int
            Nombre string
            Email  string
        }
        err := db.QueryRow(
            "SELECT id, nombre, email FROM usuarios WHERE id = $1", id,
        ).Scan(&usuario.ID, &usuario.Nombre, &usuario.Email)
        if err == sql.ErrNoRows {
            c.JSON(http.StatusNotFound, gin.H{"error": "usuario no existe"})
            return
        }
        c.JSON(http.StatusOK, usuario)
    })

    // DELETE
    r.DELETE("/usuarios/:id", func(c *gin.Context) {
        id, _ := strconv.Atoi(c.Param("id"))
        res, err := db.Exec("DELETE FROM usuarios WHERE id = $1", id)
        if err != nil {
            c.JSON(500, gin.H{"error": err.Error()})
            return
        }
        afectadas, _ := res.RowsAffected()
        if afectadas == 0 {
            c.JSON(http.StatusNotFound, gin.H{"error": "no existe"})
            return
        }
        c.Status(http.StatusNoContent)
    })

    r.Run(":8080")
}

Ejemplo con GORM

// db ya configurado con gorm.Open
usuarios := []models.Usuario{}
db.Find(&usuarios)                    // SELECT * FROM usuarios

db.Create(&models.Usuario{Nombre: "Ana", Email: "ana@mail.com"}) // INSERT

usuario := models.Usuario{}
db.First(&usuario, 1)                 // SELECT ... WHERE id = 1

usuario.Nombre = "Ana María"
db.Save(&usuario)                      // UPDATE

db.Delete(&usuario)                    // DELETE (soft delete si usar gorm.Model)

7.5 Patrones de diseño

Patrón Repository

Separa la lógica de acceso a datos del handler. Facilita testear y cambiar de base de datos:

package repos

// Interfaz del repositorio
type UsuarioRepo interface {
    Obtener(id int) (*models.Usuario, error)
    Listar() ([]models.Usuario, error)
    Crear(u *models.Usuario) error
}

// Implementación con GORM
type UsuarioGorm struct {
    db *gorm.DB
}

func NewUsuarioGorm(db *gorm.DB) *UsuarioGorm {
    return &UsuarioGorm{db: db}
}

func (r *UsuarioGorm) Obtener(id int) (*models.Usuario, error) {
    var u models.Usuario
    err := r.db.First(&u, id).Error
    return &u, err
}

func (r *UsuarioGorm) Crear(u *models.Usuario) error {
    return r.db.Create(u).Error
}

Capa de servicio

package services

type UsuarioService struct {
    repo repos.UsuarioRepo
}

func NewUsuarioService(repo repos.UsuarioRepo) *UsuarioService {
    return &UsuarioService{repo: repo}
}

func (s *UsuarioService) Obtener(id int) (*models.Usuario, error) {
    return s.repo.Obtener(id)
}

Inyección en handlers

handler := handlers.NuevoUsuarioHandler(service)
r.GET("/usuarios/:id", handler.Obtener)

Beneficios

  • Testabilidad: Puedes simular el repositorio con mocks
  • Separación de responsabilidades: Cada capa hace una sola cosa
  • Flexibilidad: Cambiar de SQL a GORM no afecta los handlers
  • Mantenibilidad: Código más ordenado y predecible

Resumen

**Lo que aprendiste:** - Los datos en memoria (maps/slices) son ideales para prototipos y cachés - database/sql ofrece una interfaz unificada para bases SQL - GORM es el ORM más popular y mapea tablas a structs de Go - Los tags de GORM definen restricciones, índices y relaciones - Los handlers se integran con consultas parametrizadas para evitar SQL injection - El patrón Repository separa datos y lógica para mejor testabilidad
