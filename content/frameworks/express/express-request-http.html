5.1 Manejo de req y res

Objetos req y res

Cada handler de ruta recibe dos objetos principales: la petición (req) y la respuesta (res). Puedes añadir un tercer parámetro next para continuar con el siguiente middleware.

Propiedades útiles de req

PropiedadDescripción
req.methodVerbo HTTP (GET, POST, etc.)
req.urlURL solicitada
req.paramsParámetros de ruta
req.queryQuery string
req.bodyCuerpo de la petición
req.headersCabeceras HTTP

Ejemplo

const express = require('express');
const app = express();

app.get('/info', (req, res) => {
  console.log('Método:', req.method);
  console.log('URL:', req.url);
  console.log('Cabeceras:', req.headers['user-agent']);
  res.send('Información de la petición registrada');
});

app.listen(3000);

---

5.2 Cuerpo de la solicitud (req.body)

Leer datos enviados por el cliente

El cuerpo de la petición contiene los datos que el cliente envía (generalmente en POST o PUT). Para leerlo en JSON debes usar el middleware express.json().

const express = require('express');
const app = express();

app.use(express.json());

app.post('/usuarios', (req, res) => {
  const { nombre, email } = req.body;

  if (!nombre || !email) {
    return res.status(400).json({ error: 'Faltan datos' });
  }

  res.status(201).json({
    mensaje: 'Usuario creado',
    usuario: { nombre, email }
  });
});

app.listen(3000);
req.body solo está disponible si se registró antes un middleware que lo analice, como express.json() o express.urlencoded().

5.3 Parámetros de ruta y query combinados

Combinar parámetros

En una misma ruta puedes leer parámetros de ruta y de query string para construir respuestas flexibles.

const express = require('express');
const app = express();

// URL: /reportes/2024/enero?formato=pdf&idioma=es
app.get('/reportes/:anio/:mes', (req, res) => {
  const { anio, mes } = req.params;
  const { formato = 'html', idioma = 'es' } = req.query;

  res.json({
    reporte: `${mes}/${anio}`,
    formato,
    idioma
  });
});

app.listen(3000);

Uso de valores por defecto

Al desestructurar req.query puedes asignar valores por defecto con el operador =, de modo que si el cliente no los envía, se usa el valor predeterminado.

Esta técnica es habitual para paginación: const pagina = Number(req.query.pagina) || 1.

5.4 Códigos de estado detallados

Usar res.status()

Para indicar el resultado de la operación, usa res.status(codigo) encadenado con un método de respuesta.

const express = require('express');
const app = express();

app.use(express.json());

app.get('/recursos/:id', (req, res) => {
  const encontrado = req.params.id === '42';

  if (!encontrado) {
    return res.status(404).json({ error: 'Recurso no encontrado' });
  }

  res.status(200).json({ id: 42, nombre: 'Recurso especial' });
});

app.listen(3000);

Grupos de códigos

GrupoRangoSignificado
Informativos100-199Información provisional
Éxito200-299La petición se completó
Redirección300-399Atención adicional requerida
Error del cliente400-499Error en la petición
Error del servidor500-599Error interno del servidor

5.5 Cabeceras (headers)

Leer y enviar cabeceras

Las cabeceras HTTP transportan metadatos de la petición y la respuesta. Se leen con req.get() o req.headers y se envían con res.set().

const express = require('express');
const app = express();

app.get('/version', (req, res) => {
  // Leer una cabecera específica
  const usuarioAgente = req.get('User-Agent');

  // Establecer cabeceras en la respuesta
  res.set('X-Version', '1.0.0');
  res.set('Cache-Control', 'no-store');

  res.json({
    mensaje: 'Cabeceras',
    tuAgente: usuarioAgente
  });
});

app.listen(3000);

Cabeceras comunes

CabeceraFunción
Content-TypeFormato de la respuesta (application/json)
AuthorizationCredenciales o token
User-AgentNavegador o cliente que hace la petición
X-Custom-*Cabeceras personalizadas de la aplicación
Por convención, las cabeceras personalizadas se nombran con el prefijo X- y usan mayúsculas separadas por guiones.

5.6 Ejemplo completo de manejo HTTP

API de tareas con métodos HTTP

const express = require('express');
const app = express();

app.use(express.json());

let tareas = [];
let idActual = 1;

app.get('/tareas', (req, res) => {
  res.status(200).json(tareas);
});

app.get('/tareas/:id', (req, res) => {
  const tarea = tareas.find(t => t.id === Number(req.params.id));
  if (!tarea) {
    return res.status(404).json({ error: 'Tarea no encontrada' });
  }
  res.status(200).json(tarea);
});

app.post('/tareas', (req, res) => {
  const { titulo } = req.body;
  if (!titulo) {
    return res.status(400).json({ error: 'El título es obligatorio' });
  }
  const nueva = { id: idActual++, titulo, completada: false };
  tareas.push(nueva);
  res.status(201).json(nueva);
});

app.put('/tareas/:id', (req, res) => {
  const tarea = tareas.find(t => t.id === Number(req.params.id));
  if (!tarea) {
    return res.status(404).json({ error: 'Tarea no encontrada' });
  }
  const { titulo, completada } = req.body;
  if (titulo !== undefined) tarea.titulo = titulo;
  if (completada !== undefined) tarea.completada = completada;
  res.status(200).json(tarea);
});

app.delete('/tareas/:id', (req, res) => {
  const index = tareas.findIndex(t => t.id === Number(req.params.id));
  if (index === -1) {
    return res.status(404).json({ error: 'Tarea no encontrada' });
  }
  tareas.splice(index, 1);
  res.status(204).end();
});

app.listen(3000);

Resumen

**Lo que aprendiste en esta sección:** - req contiene la información de la petición y res envía la respuesta - req.body lee el cuerpo de la petición usando express.json() - Se pueden combinar parámetros de ruta y query - res.status() permite devolver códigos de estado concretos - Las cabeceras se leen con req.get() y se envían con res.set() - Un ejemplo CRUD completo combina todos estos conceptos
