9.1 Qué es la autenticación

Autenticación vs autorización

La autenticación verifica la identidad de un usuario (quién eres), mientras que la autorización determina qué puede hacer (qué puedes hacer).

Enfoques comunes

EnfoqueCómo funcionaUso
SesionesEl servidor guarda el estado en cookiesApps web tradicionales
Tokens JWTToken firmado que viaja en cada peticiónAPIs y SPA
OAuthDelegar login a terceros (Google, GitHub)Login social
Para APIs REST y aplicaciones modernas, los tokens JWT son el estándar más habitual por su naturaleza sin estado.

9.2 JWT (concepto)

Qué es JWT

JWT (JSON Web Token) es un estándar para transmitir información de forma segura entre cliente y servidor. Un token está formado por tres partes codificadas en Base64 separadas por puntos: cabecera.payload.firma.

Estructura de un JWT

cabecera.payload.firma
eyJhbGciOiJIUzI1NiJ9.eyJzdWIiOiIxIn0.seK2w...
ParteContiene
CabeceraAlgoritmo de firma y tipo de token
PayloadDatos (claims) como el id del usuario
FirmaHash que garantiza que no fue alterado

Generar y verificar con jsonwebtoken

npm install jsonwebtoken
const jwt = require('jsonwebtoken');

const SECRETO = process.env.JWT_SECRET;

// Firmar un token
const token = jwt.sign(
  { id: 1, rol: 'admin' },
  SECRETO,
  { expiresIn: '1h' }
);

// Verificar el token
jwt.verify(token, SECRETO, (err, payload) => {
  if (err) return console.error('Token inválido');
  console.log('Usuario:', payload);
});
La clave secreta (SECRETO) debe almacenarse en variables de entorno y nunca en el código.

9.3 bcrypt para contraseñas

Por qué no guardar contraseñas en texto plano

Nunca almacenes contraseñas directamente. Debes aplicar un hash (función unidireccional) para que no sean legibles ni siquiera si la base de datos se ve comprometida.

npm install bcryptjs
const bcrypt = require('bcryptjs');

async function registrar(nombre, contrasena) {
  const salt = await bcrypt.genSalt(10);
  const hash = await bcrypt.hash(contrasena, salt);
  // Guardar 'hash' en la base de datos, nunca 'contrasena'
  return { nombre, hash };
}

async function verificarContrasena(contrasena, hashGuardado) {
  const coincide = await bcrypt.compare(contrasena, hashGuardado);
  return coincide; // true o false
}

Flujo de login

app.post('/login', async (req, res) => {
  const { email, contrasena } = req.body;

  // 1. Buscar usuario en BD
  // 2. Comparar la contraseña
  const coincide = await bcrypt.compare(contrasena, usuario.hash);
  if (!coincide) {
    return res.status(401).json({ error: 'Credenciales inválidas' });
  }

  // 3. Generar token JWT
  const token = jwt.sign({ id: usuario.id }, SECRETO, { expiresIn: '1h' });
  res.json({ token });
});

9.4 Manejo de sesiones o tokens

Proteger rutas con un middleware de autenticación

Puedes crear un middleware que verifique el JWT en cada petición protegida.

// middleware/auth.js
const jwt = require('jsonwebtoken');

function proteger(req, res, next) {
  const cabecera = req.headers['authorization'];
  if (!cabecera) {
    return res.status(401).json({ error: 'Token requerido' });
  }

  const token = cabecera.split(' ')[1]; // 'Bearer token'
  try {
    const payload = jwt.verify(token, process.env.JWT_SECRET);
    req.usuario = payload;
    next();
  } catch (error) {
    return res.status(401).json({ error: 'Token inválido o expirado' });
  }
}

module.exports = proteger;

Aplicar a una ruta

const proteger = require('./middleware/auth');

app.get('/perfil', proteger, (req, res) => {
  res.json({ mensaje: 'Datos del perfil', usuario: req.usuario });
});
Con este patrón, cualquier ruta protegida exige un token válido. El middleware se puede reutilizar en todas las rutas que lo necesiten.

9.5 Subida de archivos con Multer (concepto)

Qué es Multer

Multer es un middleware de Express para gestionar la subida de archivos. Procesa las peticiones de tipo multipart/form-data y guarda los archivos en disco o en memoria.

npm install multer
const multer = require('multer');
const path = require('path');

// Configurar destino y nombre de archivo
const storage = multer.diskStorage({
  destination: (req, file, cb) => cb(null, 'uploads/'),
  filename: (req, file, cb) => {
    const ext = path.extname(file.originalname);
    cb(null, Date.now() + ext);
  }
});

const upload = multer({ storage });

// Ruta que recibe un archivo en el campo 'archivo'
app.post('/subir', upload.single('archivo'), (req, res) => {
  if (!req.file) {
    return res.status(400).json({ error: 'No se subió ningún archivo' });
  }
  res.json({ mensaje: 'Archivo subido', nombre: req.file.filename });
});

Límites y validación

const upload = multer({
  storage,
  limits: { fileSize: 2 * 1024 * 1024 }, // 2 MB
  fileFilter: (req, file, cb) => {
    const permitidos = ['.png', '.jpg', '.jpeg'];
    const ext = path.extname(file.originalname).toLowerCase();
    if (!permitidos.includes(ext)) {
      return cb(new Error('Tipo de archivo no permitido'));
    }
    cb(null, true);
  }
});
Siempre valida el tamaño y el tipo de los archivos subidos para evitar riesgos de seguridad.

9.6 Ejemplo completo de autenticación

Registro, login y ruta protegida

const express = require('express');
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');
const app = express();

app.use(express.json());

const SECRETO = process.env.JWT_SECRET || 'clave-secreta';
const usuarios = []; // simula la base de datos

// Registro
app.post('/registro', async (req, res) => {
  const { email, contrasena } = req.body;
  if (!email || !contrasena) {
    return res.status(400).json({ error: 'Faltan datos' });
  }
  const existe = usuarios.find(u => u.email === email);
  if (existe) {
    return res.status(409).json({ error: 'Email ya registrado' });
  }
  const hash = await bcrypt.hash(contrasena, 10);
  const usuario = { id: usuarios.length + 1, email, hash };
  usuarios.push(usuario);
  res.status(201).json({ mensaje: 'Usuario registrado' });
});

// Login
app.post('/login', async (req, res) => {
  const { email, contrasena } = req.body;
  const usuario = usuarios.find(u => u.email === email);
  if (!usuario) {
    return res.status(401).json({ error: 'Credenciales inválidas' });
  }
  const valida = await bcrypt.compare(contrasena, usuario.hash);
  if (!valida) {
    return res.status(401).json({ error: 'Credenciales inválidas' });
  }
  const token = jwt.sign({ id: usuario.id, email }, SECRETO, { expiresIn: '1h' });
  res.json({ token });
});

// Middleware de protección
function proteger(req, res, next) {
  const cabecera = req.headers['authorization'];
  if (!cabecera) return res.status(401).json({ error: 'Token requerido' });
  try {
    req.usuario = jwt.verify(cabecera.split(' ')[1], SECRETO);
    next();
  } catch {
    res.status(401).json({ error: 'Token inválido' });
  }
}

// Ruta protegida
app.get('/perfil', proteger, (req, res) => {
  res.json({ mensaje: 'Acceso concedido', usuario: req.usuario });
});

app.listen(3000);

Resumen

**Lo que aprendiste en esta sección:** - La autenticación verifica identidad; la autorización define permisos - JWT es un token firmado que viaja en cada petición sin estado - bcrypt genera y verifica hashes seguros de contraseñas - Un middleware reusable protege las rutas con tokens - Multer gestiona la subida de archivos con límites y validación - Un flujo registro/login mayor une todos estos conceptos
