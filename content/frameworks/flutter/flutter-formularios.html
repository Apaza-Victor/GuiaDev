1.1 TextField

¿Qué es TextField?

TextField es el widget base para la entrada de texto en Flutter. Permite al usuario escribir texto en un campo.

import 'package:flutter/material.dart';

class CampoTexto extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return TextField(
      decoration: InputDecoration(
        labelText: 'Nombre',
        hintText: 'Escribe tu nombre aquí',
        prefixIcon: Icon(Icons.person),
        border: OutlineInputBorder(
          borderRadius: BorderRadius.circular(8),
        ),
      ),
    );
  }
}

Propiedades de InputDecoration

PropiedadDescripción
labelTextEtiqueta flotante
hintTextTexto de sugerencia
prefixIconIcono al inicio
suffixIconIcono al final
borderEstilo del borde
helperTextTexto de ayuda explicativo

1.2 TextEditingController

¿Qué es TextEditingController?

TextEditingController es un controlador que se vincula a un TextField para leer y manipular el texto programáticamente. Es la forma recomendada de acceder al contenido del campo.

import 'package:flutter/material.dart';

class FormularioConControlador extends StatefulWidget {
  @override
  _FormularioConControladorState createState() => _FormularioConControladorState();
}

class _FormularioConControladorState extends State<FormularioConControlador> {
  final TextEditingController nombreController = TextEditingController();
  final TextEditingController emailController = TextEditingController();

  @override
  void dispose() {
    // Liberar los controladores
    nombreController.dispose();
    emailController.dispose();
    super.dispose();
  }

  void _obtenerDatos() {
    final nombre = nombreController.text;
    final email = emailController.text;
    print('Nombre: $nombre, Email: $email');
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        TextField(
          controller: nombreController,
          decoration: InputDecoration(labelText: 'Nombre'),
        ),
        TextField(
          controller: emailController,
          decoration: InputDecoration(labelText: 'Email'),
        ),
        SizedBox(height: 16),
        ElevatedButton(
          onPressed: _obtenerDatos,
          child: Text('Enviar'),
        ),
      ],
    );
  }
}

Uso de onChanged vs Controller

  • onChanged: Recibe cada cambio de texto (para búsquedas en vivo)
  • Controller: Lee el valor puntual cuando lo necesitas

1.3 Form y FormField

¿Qué es Form?

Form agrupa varios campos y permite validarlos todos a la vez. Usa un GlobalKey para acceder al estado del formulario y disparar la validación.

import 'package:flutter/material.dart';

class FormularioConForm extends StatefulWidget {
  @override
  _FormularioConFormState createState() => _FormularioConFormState();
}

class _FormularioConFormState extends State<FormularioConForm> {
  final _formKey = GlobalKey<FormState>();
  final nombreController = TextEditingController();

  void _enviar() {
    if (_formKey.currentState!.validate()) {
      // Si la validación pasa
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Formulario válido')),
      );
    }
  }

  @override
  void dispose() {
    nombreController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Form(
      key: _formKey,
      child: Column(
        children: [
          TextFormField(
            controller: nombreController,
            decoration: InputDecoration(labelText: 'Nombre'),
            validator: (valor) {
              if (valor == null || valor.isEmpty) {
                return 'El nombre es obligatorio';
              }
              return null; // Sin errores
            },
          ),
          SizedBox(height: 16),
          ElevatedButton(
            onPressed: _enviar,
            child: Text('Enviar'),
          ),
        ],
      ),
    );
  }
}

Diferencia entre TextField y TextFormField

TextFormField amplía TextField y agrega soporte para validator y se usa dentro de un Form. Es lo que necesitas para validación integrada.

1.4 Validación

Validar campos

La validación se define en la propiedad validator del TextFormField. La función devuelve un String con el mensaje de error, o null si el valor es válido.

import 'package:flutter/material.dart';

class PreValidacion {
  static String? requerido(String? valor) {
    if (valor == null || valor.trim().isEmpty) {
      return 'Este campo es obligatorio';
    }
    return null;
  }

  static String? email(String? valor) {
    if (valor == null || valor.trim().isEmpty) {
      return 'El email es obligatorio';
    }
    final expresion = RegExp(r'^[^@]+@[^@]+\.[^@]+$');
    if (!expresion.hasMatch(valor)) {
      return 'Email no válido';
    }
    return null;
  }

  static String? minimo(int longitud) {
    return (String? valor) {
      if (valor != null && valor.length < longitud) {
        return 'Debe tener al menos $longitud caracteres';
      }
      return null;
    };
  }
}

// Uso en el formulario
class FormularioRegistro extends StatelessWidget {
  final _formKey = GlobalKey<FormState>();
  final nombreController = TextEditingController();
  final emailController = TextEditingController();
  final passController = TextEditingController();

  @override
  Widget build(BuildContext context) {
    return Form(
      key: _formKey,
      child: Column(
        children: [
          TextFormField(
            controller: nombreController,
            decoration: InputDecoration(labelText: 'Nombre'),
            validator: PreValidacion.requerido,
          ),
          TextFormField(
            controller: emailController,
            decoration: InputDecoration(labelText: 'Email'),
            validator: PreValidacion.email,
          ),
          TextFormField(
            controller: passController,
            obscureText: true,
            decoration: InputDecoration(labelText: 'Contraseña'),
            validator: PreValidacion.minimo(6),
          ),
          ElevatedButton(
            onPressed: () {
              if (_formKey.currentState!.validate()) {
                // Datos válidos
              }
            },
            child: Text('Registrarse'),
          ),
        ],
      ),
    );
  }
}

Tipos de validación

TipoEjemplo
RequeridoNo puede estar vacío
FormatoEmail, teléfono, DNI
LongitudMínimo o máximo de caracteres
ComparaciónContraseña y confirmación iguales

1.5 Recoger datos

Obtener y procesar los datos del formulario

Una vez validado el formulario, puedes recoger todos los datos y crear un modelo con ellos, enviarlos a una API o mostrarlos.

import 'package:flutter/material.dart';

// Modelo de usuario
class UsuarioRegistro {
  final String nombre;
  final String email;
  final String password;

  UsuarioRegistro({required this.nombre, required this.email, required this.password});

  Map<String, dynamic> toJson() {
    return {
      'nombre': nombre,
      'email': email,
      'password': password,
    };
  }
}

class FormularioCompleto extends StatefulWidget {
  @override
  _FormularioCompletoState createState() => _FormularioCompletoState();
}

class _FormularioCompletoState extends State<FormularioCompleto> {
  final _formKey = GlobalKey<FormState>();
  final nombreController = TextEditingController();
  final emailController = TextEditingController();
  final passController = TextEditingController();
  final confirmPassController = TextEditingController();

  bool _cargando = false;

  Future<void> _registrar() async {
    if (!_formKey.currentState!.validate()) {
      return; // No proceder si hay errores
    }

    setState(() {
      _cargando = true;
    });

    try {
      final usuario = UsuarioRegistro(
        nombre: nombreController.text,
        email: emailController.text,
        password: passController.text,
      );

      // Aquí enviarías a la API
      await Future.delayed(Duration(seconds: 2));

      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Registro exitoso: ${usuario.toJson()}')),
        );
      }
    } finally {
      if (mounted) {
        setState(() {
          _cargando = false;
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Form(
      key: _formKey,
      child: Column(
        children: [
          TextFormField(
            controller: nombreController,
            decoration: InputDecoration(labelText: 'Nombre'),
            validator: (v) => (v == null || v.isEmpty) ? 'Obligatorio' : null,
          ),
          TextFormField(
            controller: emailController,
            decoration: InputDecoration(labelText: 'Email'),
            validator: (v) => (v == null || !v.contains('@')) ? 'Email no válido' : null,
          ),
          TextFormField(
            controller: passController,
            obscureText: true,
            decoration: InputDecoration(labelText: 'Contraseña'),
            validator: (v) => (v == null || v.length < 6) ? 'Mínimo 6 caracteres' : null,
          ),
          TextFormField(
            controller: confirmPassController,
            obscureText: true,
            decoration: InputDecoration(labelText: 'Confirmar contraseña'),
            validator: (v) => (v != passController.text) ? 'No coinciden' : null,
          ),
          SizedBox(height: 16),
          _cargando
              ? CircularProgressIndicator()
              : ElevatedButton(
                  onPressed: _registrar,
                  child: Text('Registrar'),
                ),
        ],
      ),
    );
  }
}

1.6 Tipos de entrada

Otros widgets de entrada

Además de texto, Flutter ofrece widgets para otros tipos de entrada de datos.

WidgetUso
CheckboxSelección múltiple
RadioSelección única
SwitchEncendido/apagado
SliderRango numérico
DropdownButtonMenú desplegable
DatePickerSelección de fecha

Ejemplo: Entrada variada

class EntradaVariada extends StatefulWidget {
  @override
  _EntradaVariadaState createState() => _EntradaVariadaState();
}

class _EntradaVariadaState extends State<EntradaVariada> {
  bool aceptaTerminos = false;
  String genero = 'Masculino';
  int edad = 25;
  String pais = 'España';

  final List<String> paises = ['España', 'México', 'Argentina', 'Colombia'];

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        CheckboxListTile(
          title: Text('Acepto los términos'),
          value: aceptaTerminos,
          onChanged: (valor) {
            setState(() {
              aceptaTerminos = valor ?? false;
            });
          },
        ),
        ListTile(
          title: Text('Edad: $edad'),
          trailing: Slider(
            value: edad.toDouble(),
            min: 18,
            max: 65,
            onChanged: (valor) {
              setState(() {
                edad = valor.round();
              });
            },
          ),
        ),
        DropdownButtonFormField(
          value: pais,
          items: paises.map((p) {
            return DropdownMenuItem(value: p, child: Text(p));
          }).toList(),
          onChanged: (valor) {
            setState(() {
              pais = valor.toString();
            });
          },
        ),
      ],
    );
  }
}
Configura keyboardType y inputFormatters en el TextField para limitar la entrada (números, email, etc.).

Resumen

**Lo que aprendiste en esta sección:** - TextField es el widget base para entrada de texto - TextEditingController te da acceso programático al texto - Form agrupa campos y FormField valida automáticamente - La validación usa la propiedad validator que devuelve error o null - Puedes recoger todos los datos validados al enviar - Existen widgets para checkbox, radio, switch, slider y dropdown
