10.2 Entidades del proyecto

Entidad Usuario

// usuario.entity.ts
import { Entity, PrimaryGeneratedColumn, Column, OneToMany } from 'typeorm';
import { Tarea } from '../tareas/tarea.entity';

@Entity('usuarios')
export class Usuario {
  @PrimaryGeneratedColumn()
  id: number;

  @Column({ unique: true })
  email: string;

  @Column()
  password: string;

  @OneToMany(() => Tarea, (tarea) => tarea.usuario)
  tareas: Tarea[];
}

Entidad Tarea

// tarea.entity.ts
import { Entity, PrimaryGeneratedColumn, Column, ManyToOne, JoinColumn } from 'typeorm';
import { Usuario } from '../usuarios/usuario.entity';

@Entity('tareas')
export class Tarea {
  @PrimaryGeneratedColumn()
  id: number;

  @Column()
  titulo: string;

  @Column({ default: false })
  completada: boolean;

  @ManyToOne(() => Usuario, (usuario) => usuario.tareas)
  @JoinColumn({ name: 'usuario_id' })
  usuario: Usuario;

  @Column({ name: 'usuario_id' })
  usuarioId: number;
}

10.3 Autenticación integrada

Servicio de autenticación

import { Injectable, UnauthorizedException } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import * as bcrypt from 'bcrypt';
import { Usuario } from '../usuarios/usuario.entity';

@Injectable()
export class AuthService {
  constructor(
    @InjectRepository(Usuario)
    private readonly usuarios: Repository<Usuario>,
    private readonly jwtService: JwtService,
  ) {}

  async registrar(email: string, password: string) {
    const encriptada = await bcrypt.hash(password, 10);
    const usuario = this.usuarios.create({ email, password: encriptada });
    return this.usuarios.save(usuario);
  }

  async login(email: string, password: string) {
    const usuario = await this.usuarios.findOne({ where: { email } });
    if (!usuario || !(await bcrypt.compare(password, usuario.password))) {
      throw new UnauthorizedException('Credenciales inválidas');
    }
    const payload = { sub: usuario.id, email: usuario.email };
    return { access_token: this.jwtService.sign(payload) };
  }
}

Controlador en /auth

@Controller('auth')
export class AuthController {
  constructor(private readonly authService: AuthService) {}

  @Post('registro')
  registrar(@Body() dto: RegistrarDto) {
    return this.authService.registrar(dto.email, dto.password);
  }

  @Post('login')
  login(@Body() dto: LoginDto) {
    return this.authService.login(dto.email, dto.password);
  }
}

10.4 CRUD de tareas protegido

Servicio de tareas vinculado al usuario

import { Injectable, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Tarea } from './tarea.entity';

@Injectable()
export class TareasService {
  constructor(
    @InjectRepository(Tarea)
    private readonly tareas: Repository<Tarea>,
  ) {}

  listar(usuarioId: number): Promise<Tarea[]> {
    return this.tareas.find({ where: { usuarioId } });
  }

  crear(usuarioId: number, titulo: string): Promise<Tarea> {
    return this.tareas.save({ usuarioId, titulo });
  }

  async actualizar(usuarioId: number, id: number, datos: Partial<Tarea>) {
    const tarea = await this.tareas.findOne({ where: { id, usuarioId } });
    if (!tarea) throw new NotFoundException('Tarea no encontrada');
    Object.assign(tarea, datos);
    return this.tareas.save(tarea);
  }

  async eliminar(usuarioId: number, id: number) {
    const tarea = await this.tareas.findOne({ where: { id, usuarioId } });
    if (!tarea) throw new NotFoundException('Tarea no encontrada');
    await this.tareas.remove(tarea);
  }
}

Controlador protegido

import { Controller, Get, Post, Put, Delete, Body, Param, ParseIntPipe, UseGuards, Request } from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { TareasService } from './tareas.service';

@Controller('tareas')
@UseGuards(AuthGuard('jwt'))
export class TareasController {
  constructor(private readonly tareasService: TareasService) {}

  @Get()
  listar(@Request() req) {
    return this.tareasService.listar(req.user.userId);
  }

  @Post()
  crear(@Request() req, @Body('titulo') titulo: string) {
    return this.tareasService.crear(req.user.userId, titulo);
  }

  @Put(':id')
  actualizar(@Request() req, @Param('id', ParseIntPipe) id: number, @Body() datos: any) {
    return this.tareasService.actualizar(req.user.userId, id, datos);
  }

  @Delete(':id')
  eliminar(@Request() req, @Param('id', ParseIntPipe) id: number) {
    return this.tareasService.eliminar(req.user.userId, id);
  }
}

10.5 Despliegue en producción

Compilar para producción

# Compilar a la carpeta dist
npm run build

# Ejecutar la versión compilada
npm run start:prod

# Ejecutar directamente desde dist
node dist/main

Variables de entorno

# .env de producción
PORT=8080
JWT_SECRET=clave_produccion_muy_larga
DB_HOST=db.internal.com
DB_PORT=5432
DB_USER=app_user
DB_PASSWORD=segura

Cargar variables con @nestjs/config

import { Module } from '@nestjs/common';
import { ConfigModule } from '@nestjs/config';

@Module({
  imports: [
    ConfigModule.forRoot({
      isGlobal: true,
    }),
  ],
})
export class AppModule {}

Usar la configuración

const port = process.env.PORT ?? 3000;
await app.listen(port);
En producción desactiva `synchronize: true` y usa migraciones. Asegúrate de que el JWT_SECRET sea largo, aleatorio y nunca se suba al repositorio.

10.6 Buenas prácticas y despliegue en la nube

Buenas prácticas

  • Usar DTOs y ValidationPipe para validar toda la entrada
  • Proteger todas las rutas privadas con guards de autenticación
  • No exponer errores internos: registrar y devolver mensajes genéricos
  • Usar variables de entorno para secretos y configuración
  • Escribir pruebas con Jest (unitarias y e2e)
  • Aplicar migraciones en lugar de sincronización automática

Plataformas de despliegue comunes

PlataformaCaracterística
RenderDespliegue sencillo de servicios y bases de datos
RailwayIntegración con repositorios y variables de entorno
Fly.ioContenedores distribuidos a nivel global
Docker + VPSMáximo control sobre el entorno
AWS / GCP / AzureEscalado empresarial

Dockerfile de ejemplo

FROM node:20-alpine AS build
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
RUN npm run build

FROM node:20-alpine
WORKDIR /app
COPY --from=build /app/dist ./dist
COPY --from=build /app/package*.json ./
RUN npm install --omit=dev
EXPOSE 3000
CMD ["node", "dist/main"]
Antes de desplegar, prueba localmente con `npm run build` y `npm run start:prod` para detectar errores de compilación o de dependencias omitidas.
