Módulos core: Foundation y Swift estándar Base

import Foundation

// La stdlib de Swift + Foundation de Apple: todo lo esencial
let nums = [3, 1, 4, 1, 5]
let ordenados = nums.sorted()
let pares = nums.filter { $0 % 2 == 0 }.map { $0 * 10 }

// Dictionaries y sets
var frec: [String: Int] = [:]
"hola mundo hola".split(separator: " ").forEach { frec[String($0), default: 0] += 1 }

// Fechas y JSON (Codable)
struct Usuario: Codable {
    let nombre: String
    let edad: Int
}

let json = #"{"nombre":"Ana","edad":30}"#
let data = json.data(using: .utf8)!
let usuario = try! JSONDecoder().decode(Usuario.self, from: data)

// Strings con interpolación
let saludo = "Hola \(usuario.nombre), edad \(usuario.edad)"

La stdlib de Swift (+ Foundation/Codable) cubren colecciones, JSON, fechas y red sin dependencias. `Codable` serializa/deserializa seguro y tipo-verificado.

Swift Package Manager (SPM) Deps

# SPM: el gestor de paquetes de Swift (similar a Cargo)
swift package init --type executable
swift build
swift run
swift test
npm styled: 'swift package add' no existe; se edita Package.swift

# Package.swift
// swift-tools-version:5.9
import PackageDescription

let package = Package(
    name: "MiApp",
    dependencies: [
        .package(url: "https://github.com/vapor/vapor.git", from: "4.0.0")
    ],
    targets: [
        .executableTarget(name: "MiApp", dependencies: ["Vapor"])
    ]
)

SPM descarga desde git (GitHub) o registros; resolve + lockfile `Package.resolved`. Es el equivalente a Cargo/npm dentro del ecosistema Swift.

async/await y Combine Async

import Foundation

// async/await nativo desde Swift 5.5
func fetchUsuarios() async throws -> [Usuario] {
    let (data, _) = try await URLSession.shared.data(
        from: URL(string: "https://api.example.com/usuarios")!
    )
    return try JSONDecoder().decode([Usuario].self, from: data)
}

func main() async {
    do {
        let usuarios = try await fetchUsuarios()
        for u in usuarios { print(u.nombre) }
    } catch {
        print("error: \(error)")
    }
}
// async let para paralelismo sencillo
// TaskGroup para fan-out/fan-in

// Combine: framework de reactivo de Apple (publicadores/subscriptores)
// Foundation también: @Observable / Observation (iOS 17+)

async/await + URLSession + Codable son el patrón estándar de red. `@Observable` y el observation framework modernizan el estado reactivo en la UI.

Instalación de Swift Inicio

# macOS: Xcode / Command Line Tools traen Swift completo
xcode-select --install

# Linux (Ubuntu/Debian) y Windows (Swift 6+) descargar el toolchain
# download.swift.org →  .tar.gz (Linux) o .exe (Windows)

# Verificar
swift --version
echo 'print("Hola desde Swift")' > hola.swift
swift hola.swift

# En macOS/iOS usarás Xcode (IDE) + SwiftUI/UIKit

El ecosistema iOS (Xcode, Simulator) solo funciona en macOS. Swift de servidor (Vapor, Hummingbird) y Linux/Windows para backend sí son multiplataforma.
