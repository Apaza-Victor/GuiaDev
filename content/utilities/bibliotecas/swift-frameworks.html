Para apps Apple: SwiftUI+Combine y sin más. Para backend: Vapor. Compartir modelos Codable entre tu app y tu API es gratis porque ambos son Swift.

SwiftUI y Combine UI

import SwiftUI

struct ContadorView: View {
    @State private var cuenta = 0
    @Environment(\.colorScheme) var esquema

    var body: some View {
        VStack(spacing: 20) {
            Text("Cuenta: \(cuenta)")
                .font(.largeTitle.bold())
            HStack {
                Button("+") { cuenta += 1 }
                Button("−") { cuenta -= 1 }
            }
        }
        .padding()
    }
}

// Combine: @State/@ObservableObject, @Published, .sink
import Combine
final class VM: ObservableObject {
    @Published var items: [String] = []
}

struct MiApp: App {
    var body: some Scene { WindowGroup { ContadorView() } }
}

SwiftUI es declarativo y es el mantenedor de la UI moderno (iOS 13+). Combine o el nuevo Observation gestionan estado reactivo con `@Published`.

Vapor: servidor Swift Backend

# Crear proyecto Vapor
curl -sL https://templates.vapor.cloud/template.tar.gz | tar -xz
# o app generator: vapor new mi-api

import Vapor

func routes(_ app: Application) throws {
    app.get { req in
        "Bienvenido a mi API Swift"
    }
    app.get("saludo", ":nombre") { req -> String in
        let nombre = try req.parameters.require("nombre")
        return "Hola \(nombre)"
    }
    app.post("usuarios") { req in
        let u = try req.content.decode(Usuario.self)
        return u
    }
}

@main
struct AppMain {
    static func main() async throws {
        var env = try Environment.detect()
        let app = Application(env)
        defer { app.shutdown() }
        try routes(app)
        try await app.execute()
    }
}

Vapor trae routing, middlewares, Fluent (ORM async), WebSockets y corutinas async/await. Hummingbird es la alternativa ligera. `req.content.decode` con Codable.

Fluent y persistencia Datos

import Fluent
import FluentSQLiteDriver

final class Usuario: Model {
    static let schema = "usuarios"
    @ID(key: .id) var id: UUID?
    @Field(key: "nombre") var nombre: String
    @Field(key: "edad") var edad: Int
    @Timestamp(key: "creado_en", on: .create) var creadoEn: Date?
}

// Migraciones + SQLiteDriver
final class CrearUsuario: AsyncMigration {
    func prepare(on db: Database) async throws {
        try await db.schema("usuarios")
            .id()
            .field("nombre", .string, .required)
            .field("edad", .int, .required)
            .create()
    }
    func revert(on db: Database) async throws {
        try await db.schema("usuarios").delete()
    }
}

// En ruta: Usuario.query(on: req.db).filter(\.$edad > 18).all()

Fluent es el ORM de Vapor (async, multiplataforma: SQLite/Postgres/MySQL). `@Field/@ID/@Timestamp` decoran el modelo y las migraciones controlan el esquema.

Testing con XCTest Calidad

import XCTest

final class MiAppTests: XCTestCase {
    override func setUp() async throws { super.setUp() }

    func testSuma() {
        XCTAssertEqual(suma(2, 3), 5)
    }

    func testJSON() throws {
        let json = #"{"nombre":"Ana","edad":30}"#
        let u = try JSONDecoder().decode(Usuario.self,
            from: json.data(using: .utf8)!)
        XCTAssertEqual(u.nombre, "Ana")
    }
}
// swift test (SPM) o ⌘U en Xcode
// Cobertura: Xcode Coverage (esquema de test) / Codecov
// UI tests: XCUITest en Xcode para flows de la app

XCTest es el framework nativo (unit + UI). `swift test` lo corre desde CLI; Xcode muestra cobertura. Para APIs de Vapor puedes testear con método `app.test()`.
