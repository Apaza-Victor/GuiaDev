Funciones genéricas

function primero<T>(arr: T[]): T | undefined {
  return arr[0];
}

primero<number>([1, 2, 3]); // number | undefined
primero(["a", "b"]);        // string | undefined

Constraints con extends

interface ConLongitud { length: number }

function logLongitud<T extends ConLongitud>(x: T): T {
  console.log(x.length);
  return x;
}

logLongitud("hola");     // OK
logLongitud([1, 2]);     // OK
// logLongitud(123);     // Error: number no tiene length

Genéricos en interfaces

interface Respuesta<T> {
  ok: boolean;
  data: T;
}

const r: Respuesta<string[]> = {
  ok: true,
  data: ["uno", "dos"]
};
