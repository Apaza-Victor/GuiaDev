Utility Types

interface Tarea { id: number; titulo: string; done: boolean }

type Borrador  = Partial<Tarea>;       // todo opcional
type Vista     = Pick<Tarea, "titulo">;// solo titulo
type SinId     = Omit<Tarea, "id">;    // todo menos id
type MapaTareas = Record<number, Tarea>;

Narrowing

function formatear(x: string | number): string {
  if (typeof x === "string") return x.trim();  // aquí x es string
  return x.toFixed(2);                          // aquí x es number
}
Consejo: El narrowing también funciona con instanceof, in y predicados personalizados (x is Tipo).
