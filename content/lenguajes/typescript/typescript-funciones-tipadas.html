Parámetros y retorno

function enviar(mensaje: string, veces: number = 1): void {
  for (let i = 0; i < veces; i++) console.log(mensaje);
}

function total(...nums: number[]): number {
  return nums.reduce((a, b) => a + b, 0);
}

enviar("Hola", 2);   // imprime 2 veces
total(1, 2, 3);      // 6

Sobrecarga

function convertir(valor: string): number | undefined;
function convertir(valor: number): string;
function convertir(valor: any): any {
  return typeof valor === "string" ? parseInt(valor) : String(valor);
}

convertir("42"); // number | undefined
convertir(42);   // string
