Clases

class Cuenta {
private:
    double saldo_ = 0.0;
public:
    Cuenta(double inicial) : saldo_(inicial) {}   // lista de inicialización

    void depositar(double m) { saldo_ += m; }
    double saldo() const { return saldo_; }       // método const
};

Herencia y polimorfismo

class Figura {
public:
    virtual double area() const { return 0; }
    virtual ~Figura() = default;                  // destructor virtual!
};

class Circulo : public Figura {
    double r_;
public:
    explicit Circulo(double r) : r_(r) {}
    double area() const override { return 3.14159 * r_ * r_; }
};
Cuidado: Si una clase va a heredarse, declara su destructor virtual; si no, eliminar vía puntero base es comportamiento indefinido.
