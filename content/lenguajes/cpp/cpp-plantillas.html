Plantillas de función

template <typename T>
T maximo(const T& a, const T& b) {
    return (a > b) ? a : b;
}

maximo(3, 7);        // int
maximo(2.5, 1.8);    // double

Plantillas de clase

template <typename T>
class Caja {
    T valor_;
public:
    void set(const T& v) { valor_ = v; }
    T get() const { return valor_; }
};

Caja<std::string> caja;
caja.set("contenido");
