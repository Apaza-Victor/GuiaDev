Condicionales y bucles

for (int i = 0; i < 5; ++i) std::cout << i;

for (int n : {1, 2, 3}) std::cout << n;   // range-based

while (true) { break; }

if (int r = calcular(); r > 0) {  // inicializador en if (C++17)
    usar(r);
}

Funciones y referencias

void intercambiar(int& a, int& b) {   // por referencia
    int t = a; a = b; b = t;
}

int suma(int a, int b) { return a + b; }
double suma(double a, double b) { return a + b; }  // sobrecarga

auto lambda = [](int x) { return x * x; };  // lambda
