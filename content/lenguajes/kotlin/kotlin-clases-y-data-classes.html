Clases

class Usuario(val nombre: String, var edad: Int) {
    fun cumplirAnios() { edad++ }
}

val u = Usuario("Ana", 30)
u.cumplirAnios()
println(u.edad)   // 31

Data classes

data class Punto(val x: Int, val y: Int)

val a = Punto(1, 2)
val b = a.copy(x = 10)
a == Punto(1, 2)   // true: equals generado
println(a)          // Punto(x=1, y=2): toString generado

Sealed y object

sealed class Estado {
    object Cargando : Estado()
    data class Exito(val datos: List<String>) : Estado()
    data class Error(val mensaje: String) : Estado()
}

when (estado) {
    is Estado.Cargando -> mostrarSpinner()
    is Estado.Exito -> pintar(estado.datos)
    is Estado.Error -> avisar(estado.mensaje)
}
