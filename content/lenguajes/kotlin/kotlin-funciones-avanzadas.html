Default y named args

fun saludar(nombre: String, saludo: String = "Hola") =
    "$saludo, $nombre!"

saludar("Ana")                          // Hola, Ana!
saludar("Ana", saludo = "Bienvenida")   // named arg

Extension functions

fun String.gritar(): String = uppercase() + "!!!"

"hola".gritar()   // "HOLA!!!"

fun Int.doble() = this * 2
5.doble()         // 10

Lambdas

val suma = { a: Int, b: Int -> a + b }
suma(2, 3)

val numeros = listOf(1, 2, 3)
numeros.filter { it % 2 == 0 }
       .map { it * 10 }
       .forEach(::println)
