val interpolado = "Valor: $entero, suma: ${entero + 8}"

Null-safety

var nombre: String = "Ana"
// nombre = null              // error: no nullable

var apodo: String? = null     // nullable con ?

apodo?.length                 // safe call: null si apodo es null
apodo ?: "sin apodo"          // Elvis: valor por defecto

// apodo!!.length             // NPE si es null: evitar!
Consejo: El 90% de los NullPointerException desaparecen usando ?. y ?: en lugar de !!.
