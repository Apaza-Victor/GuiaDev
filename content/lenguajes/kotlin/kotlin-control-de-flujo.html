if expresión y when

val resultado = if (nota >= 70) "Aprobado" else "Reprobado"

val nivel = when {
    nota >= 90 -> "Excelente"
    nota >= 70 -> "Aprobado"
    else -> "Reprobado"
}

when (dia) {
    6, 7 -> println("finde")
    in 1..5 -> println("entre semana")
    else -> println("inválido")
}

Bucles y ranges

for (i in 1..5) print(i)          // 12345
for (i in 1 until 5) print(i)     // 1234
for (i in 5 downTo 1 step 2) print(i)  // 531

val lista = listOf("a", "b", "c")
for ((indice, valor) in lista.withIndex()) {
    println("$indice: $valor")
}
