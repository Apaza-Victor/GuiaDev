Conceptos base

import kotlinx.coroutines.*

suspend fun descargar(): String {
    delay(1000)                       // no bloquea el hilo
    return "contenido"
}

fun main() = runBlocking {
    launch {                          // fire and forget
        repeat(3) { delay(300); println("latido") }
    }
    val resultado = async { descargar() }.await()
    println(resultado)
}
Consejo: Una corrutina consume ~unos bytes frente a ~1MB de un hilo: puedes lanzar decenas de miles simultáneas.
