Operaciones funcionales

data class Persona(val nombre: String, val edad: Int)

val personas = listOf(
    Persona("Ana", 30),
    Persona("Luis", 17),
    Persona("María", 25)
)

personas.filter { it.edad >= 18 }
        .sortedBy { it.nombre }
        .map { it.nombre }
        .forEach(::println)

val totalEdad = personas.sumOf { it.edad }
val hayMenores = personas.any { it.edad < 18 }
