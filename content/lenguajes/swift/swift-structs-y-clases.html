Structs (value type)

struct Punto {
    var x = 0.0, y = 0.0

    mutating func mover(dx: Double, dy: Double) {
        x += dx; y += dy
    }
}

var a = Punto(x: 1, y: 2)
var b = a                  // COPIA independiente
b.x = 99
print(a.x)                 // 1

Clases (reference type)

class Contador {
    var total = 0
    func incrementar() { total += 1 }
}

let c = Contador()         // let pero mutable por dentro
c.incrementar()

let d = c                  // MISMA referencia
d.total                    // 1
Contador === c             // identidad
Consejo: Regla general de Apple: usa structs por defecto; clases solo cuando necesites herencia, identidad o Objective-C.
