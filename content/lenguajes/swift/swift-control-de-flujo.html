Switch avanzado

switch nota {
case 90...100: print("Excelente")
case 70..<90:  print("Aprobado")
case let x where x < 70: print("Reprobado: \(x)")
default: break
}

let punto = (1, 1)
switch punto {
case (0, 0): print("origen")
case (_, 0): print("eje X")
case (let x, let y) where x == y: print("diagonal")
default: print("cuadrante")
}
