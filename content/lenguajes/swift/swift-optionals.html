¿Qué es un optional?

var edad: Int? = nil       // puede no tener valor
edad = 25

// Fuerza (!) peligrosa: crash si es nil
// let valor = edad!

if let e = edad {           // unwrapping seguro
    print("Tiene \(e) años")
}

guard let y ??

func procesar(edad: Int?) {
    guard let edad = edad, edad >= 18 else {
        print("Edad inválida o menor de edad")
        return               // guard exige salir del scope
    }
    print("Adulto de \(edad) años")   // edad ya desenvolvida
}

let nombre: String? = nil
let final = nombre ?? "anónimo"    // valor por defecto
Consejo: Prefiere guard let cuando la ausencia del valor impide continuar: mantiene el código principal sin anidar.
