Funciones

func saludar(a nombre: String, veces: Int = 1) -> String {
    return String(repeating: "Hola \(nombre)! ", count: veces)
}

saludar(a: "Ana", veces: 2)

func dividir(_ a: Double, _ b: Double) -> (cociente: Double, resto: Double)? {
    guard b != 0 else { return nil }
    return (a / b, a.truncatingRemainder(dividingBy: b))
}

Closures

let cuadrado = { (x: Int) -> Int in x * x }
cuadrado(4)   // 16

let nums = [3, 1, 4, 1, 5]
let ordenados = nums.sorted { $0 > $1 }        // shorthand args
let pares = nums.filter { $0 % 2 == 0 }
let dobles = nums.map { $0 * 2 }
