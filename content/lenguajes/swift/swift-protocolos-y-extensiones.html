Protocolos

protocol Describible {
    var descripcion: String { get }
    func describir() -> String
}

extension Describible {
    func describir() -> String {   // implementación por defecto
        return descripcion
    }
}

struct Libro: Describible {
    let titulo: String
    var descripcion: String { "Libro: \(titulo)" }
}

Extensiones

extension Int {
    func veces(_ tarea: () -> Void) {
        for _ in 0..<self { tarea() }
    }
}

3.veces { print("hola") }
