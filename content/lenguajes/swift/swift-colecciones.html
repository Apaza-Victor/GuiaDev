Las tres colecciones

var lista = [1, 2, 3]
lista.append(4)
lista[0] = 10

var conjunto: Set = [1, 2, 2, 3]   // {1, 2, 3}
conjunto.insert(4)

var diccionario = ["ana": 30, "luis": 25]
diccionario["maría"] = 28
let edad = diccionario["ana"] ?? 0  // siempre optional al leer
