if / else

int nota = 85;

if (nota >= 90)      Console.WriteLine("Excelente");
else if (nota >= 70) Console.WriteLine("Aprobado");
else                 Console.WriteLine("Reprobado");

string resultado = nota >= 70 ? "Aprobado" : "Reprobado"; // ternario

Switch expression (moderno)

string nivel = nota switch
{
    >= 90 => "Excelente",
    >= 80 => "Muy bueno",
    >= 70 => "Aprobado",
    _     => "Reprobado"
};

Bucles

for (int i = 0; i < 5; i++) Console.Write(i);

int[] nums = { 1, 2, 3 };
foreach (int n in nums) Console.Write(n);

int j = 0;
while (j < 3) { Console.Write(j); j++; }
