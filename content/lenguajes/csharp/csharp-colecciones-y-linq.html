LINQ básico

using System.Linq;

var numeros = new[] { 5, 2, 8, 1, 9 };

var pares    = numeros.Where(n => n % 2 == 0);
var dobles   = numeros.Select(n => n * 2);
var ordenado = numeros.OrderBy(n => n);
var maximo   = numeros.Max();
var hayMayores = numeros.Any(n => n > 8);   // true

LINQ con objetos

var adultos = personas
    .Where(p => p.Edad >= 18)
    .OrderBy(p => p.Nombre)
