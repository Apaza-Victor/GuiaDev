Definición básica

static int Sumar(int a, int b)
{
    return a + b;
}

static void Saludar(string nombre = "mundo") =>
    Console.WriteLine($"Hola, {nombre}!");

Sumar(2, 3);
Saludar();               // usa el valor por defecto
Saludar(nombre: "Ana");  // parámetro nombrado

ref y out

static void Duplicar(ref int x) => x *= 2;

static bool Dividir(int a, int b, out int cociente)
{
    cociente = 0;
    if (b == 0) return false;
    cociente = a / b;
    return true;
}

int v = 5;
Duplicar(ref v);                 // v = 10
Dividir(10, 3, out int c);       // c = 3

Sobrecarga

static void Log(string msg) => Console.WriteLine(msg);
static void Log(int codigo) => Console.WriteLine($"[{codigo}]");
static void Log(params object[] datos) => Console.WriteLine(string.Join(", ", datos));
