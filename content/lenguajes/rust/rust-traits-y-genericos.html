Traits

trait Resumen {
    fn autor(&self) -> String;

    fn preview(&self) -> String {       // método con implementación
        format!("por {}", self.autor())
    }
}

struct Articulo { titulo: String }

impl Resumen for Articulo {
    fn autor(&self) -> String { String::from("Equipo GuiaDev") }
}

Genéricos con bounds

fn mayor<T: PartialOrd>(a: T, b: T) -> T {
    if a > b { a } else { b }
}

fn imprimir<T: std::fmt::Display>(item: T) {
    println!("{}", item);
}

mayor(3, 7);
imprimir("texto");
