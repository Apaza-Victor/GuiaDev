Enums con datos

enum Forma {
    Circulo(f64),
    Rectangulo(f64, f64),
}

fn area(f: &Forma) -> f64 {
    match f {
        Forma::Circulo(r) => 3.14159 * r * r,
        Forma::Rectangulo(a, b) => a * b,
    }
}

Option: sin null

fn buscar(id: u32) -> Option<String> {
    if id == 1 { Some(String::from("Ana")) } else { None }
}

match buscar(1) {
    Some(nombre) => println!("Encontrado: {}", nombre),
    None => println!("No existe"),
}

// Atajos comunes
let nombre = buscar(2).unwrap_or(String::from("anónimo"));
Consejo: Option<T> reemplaza al null: el compilador te obliga a manejar el caso vacío antes de usar el valor.
