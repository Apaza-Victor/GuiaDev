Ownership

let s1 = String::from("hola");
let s2 = s1;              // MOVE: s1 ya no es válido
// println!("{}", s1);    // error: value borrowed after move

let n1 = 5;
let n2 = n1;              // COPY (tipos simples)
println!("{} {}", n1, n2); // OK
Cuidado: Al asignar o pasar a una función, los tipos complejos (String, Vec) se mueven; los simples (int, bool) se copian.

Referencias (borrowing)

fn longitud(s: &String) -> usize {  // presta, no mueve
    s.len()
}

let s = String::from("hola");
let l = longitud(&s);
println!("'{}' mide {}", s, l);      // s sigue válida

fn agregar(s: &mut String) {
    s.push_str(", mundo");           // préstamo mutable
}

let mut t = String::from("hola");
agregar(&mut t);                     // solo UN &mut a la vez

Slices

let texto = String::from("Hola mundo");
let hola = &texto[0..4];     // "Hola"
let mundo = &texto[5..];     // "mundo"

let nums = [1, 2, 3, 4];
let mitad = &nums[0..2];     // [1, 2]
