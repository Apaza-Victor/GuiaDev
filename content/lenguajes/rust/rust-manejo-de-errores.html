Result

use std::num::ParseIntError;

fn parsear(s: &str) -> Result<i32, ParseIntError> {
    s.parse::<i32>()
}

match parsear("42") {
    Ok(n) => println!("Número: {}", n),
    Err(e) => println!("Error: {}", e),
}

El operador ?

use std::fs::File;
use std::io::{self, Read};

fn leer_config() -> Result<String, io::Error> {
    let mut s = String::new();
    File::open("config.txt")?.read_to_string(&mut s)?;
    Ok(s)   // ? propaga el error hacia arriba automáticamente
}
Consejo: Usa panic! solo para errores irrecuperables (bugs); para todo lo demás devuelve Result.
