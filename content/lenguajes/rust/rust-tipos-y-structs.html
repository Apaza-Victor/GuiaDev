Structs y métodos

struct Rectangulo {
    ancho: f64,
    alto: f64,
}

impl Rectangulo {
    fn nuevo(ancho: f64, alto: f64) -> Self {
        Self { ancho, alto }
    }

    fn area(&self) -> f64 {          // método
        self.ancho * self.alto
    }
}

let r = Rectangulo::nuevo(3.0, 4.0);
println!("{}", r.area());            // 12
