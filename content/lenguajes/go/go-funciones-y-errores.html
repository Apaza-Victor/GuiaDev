Múltiples retornos y error

func dividir(a, b float64) (float64, error) {
	if b == 0 {
		return 0, fmt.Errorf("división por cero")
	}
	return a / b, nil
}

res, err := dividir(10, 2)
if err != nil {
	log.Fatal(err)
}
fmt.Println(res)

func total(nums ...int) int {
	t := 0
	for _, n := range nums { t += n }
	return t
}

total(1, 2, 3)
