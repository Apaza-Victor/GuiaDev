Structs

type Usuario struct {
	Nombre string
	Edad   int
}

u := Usuario{Nombre: "Ana", Edad: 30}
u.Edad = 31

p := &u                  // puntero a struct
p.Nombre = "Ana María"   // acceso directo, sin ->

Métodos y composición

func (u Usuario) Saludar() string {
	return "Hola, " + u.Nombre
}

func (u *Usuario) CumplirAnios() {   // receptor puntero: muta
	u.Edad++
}

type Admin struct {
	Usuario      // embebido: composición
	Permisos []string
}

a := Admin{}
a.Saludar()   // promovido desde Usuario
