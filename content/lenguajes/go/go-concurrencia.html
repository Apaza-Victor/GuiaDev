func worker(id int, ch chan<- string) {
	time.Sleep(time.Second)
	ch <- fmt.Sprintf("worker %d listo", id)
}

ch := make(chan string)
for i := 1; i <= 3; i++ {
	go worker(i, ch)      // ligero: miles son viables
}

for i := 0; i < 3; i++ {
	fmt.Println(<-ch)     // recibe cada resultado
}
