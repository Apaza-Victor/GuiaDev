Tipos primitivos Básico

String

let nombre = "Juan";
let apellido = 'Pérez';
let mensaje = `Hola ${nombre}`; // Template literal

Number

let entero = 42;
let decimal = 3.14;
let infinito = Infinity;
let noEsNumero = NaN;

Boolean

let verdadero = true;
let falso = false;

Null y Undefined

let valorNulo = null;
let valorIndefinido = undefined;

Symbol

let simbolo = Symbol('descripción');
let simboloUnico = Symbol.for('clave');

BigInt

let numeroGrande = 9007199254740991n;
let otroBigInt = BigInt("9007199254740991");

Tipos estructurados

Objetos

let persona = {
    nombre: "María",
    edad: 25,
    saludar() {
        return `Hola, soy ${this.nombre}`;
    }
};

Arrays

let numeros = [1, 2, 3, 4, 5];
let mixto = ["texto", 42, true, null];

Verificación de tipos

typeof

typeof "Hola"          // "string"
typeof 42              // "number"
typeof true            // "boolean"
typeof undefined       // "undefined"
typeof null            // "object" (¡esto es un bug histórico!)
typeof {}              // "object"
typeof []              // "object"
typeof Symbol()        // "symbol"
typeof 42n             // "bigint"

instanceof

[] instanceof Array            // true
new Date() instanceof Date     // true
/regex/ instanceof RegExp      // true

Conversión de tipos

Conversión a String

String(123)           // "123"
(123).toString()      // "123"
123 + ""             // "123"

Conversión a Number

Number("123")        // 123
parseInt("123")      // 123
parseFloat("3.14")   // 3.14
+"123"              // 123

Conversión a Boolean

Boolean(1)           // true
Boolean("")          // false
Boolean([])          // true
!!""                // false

Valores falsy

  • false
  • 0
  • ""
  • null
  • undefined
  • NaN
if ("") {
    console.log("Esto no se ejecuta");
}

Valores truthy

  • Todo lo demás es truthy
if ("texto") {
    console.log("Esto sí se ejecuta");
}

Coerción de tipos

"5" + 3      // "53" (string)
"5" - 3      // 2 (number)
"5" * "3"    // 15 (number)
[] + {}      // "[object Object]"

Mejores prácticas

  1. Usar comparación estricta
// ❌ Malo
5 == "5"    // true

// ✅ Bueno
5 === "5"   // false
  1. Verificar tipos explícitamente
// ✅ Bueno
if (typeof variable === "string") {
    // hacer algo
}
  1. Manejar valores nulos/indefinidos
// ✅ Bueno
const valor = obtenerValor() ?? valorPorDefecto;
