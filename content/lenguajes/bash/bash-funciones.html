Definición y uso

saludar() {
    local quien="$1"          # local: no contamina el scope global
    echo "Hola, $quien"
}

saludar "Ana"                 # llamada SIN paréntesis

Exit status

existe_archivo() {
    [ -f "$1" ]
}                             # el return implícito es el último comando

if existe_archivo "/etc/passwd"; then
    echo "existe"
fi

echo $?    # 0 éxito, distinto de 0 error
Consejo: En bash las funciones devuelven status (0-255), no valores. Para "retornar" datos usa echo y captura con $(funcion).
