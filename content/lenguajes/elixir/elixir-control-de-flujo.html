case: pattern matching como flujo

case es la herramienta primaria de control de flujo: evalúa una expresión y la compara con varios patrones. No hay if/else if masivo: se modela con patrones.

defmodule Clasificador do
  def tipo_dia(dia) do
    case dia do
      "sábado" -> "fin de semana"
      "domingo" -> "fin de semana"
      "lunes" -> "laboral"
      "viernes" -> "laboral"
      _ -> "día arbitrario"
    end
  end

  def duplicar(lista) do
    case lista do
      [] -> []
      [head | tail] -> [head * 2 | duplicar(tail)]
    end
  end

  def clasificar_resultado(result) do
    case result do
      {:ok, valor} -> "éxito: #{valor}"
      {:error, motivo} -> "error: #{motivo}"
      nil -> "sin resultado"
      _ -> "resultado desconocido"
    end
  end
end

IO.inspect(Clasificador.tipo_dia("domingo"))
IO.inspect(Clasificador.duplicar([1, 2, 3]))
IO.inspect(Clasificador.clasificar_resultado({:ok, 42}))

El comodín _ captura todo. Elixir con patrón [head | tail] + recursión sustituye a los bucles. Es la forma pura de recorrer listas.

if, unless y cond

if/unless cubren condicionales simples; cond permite una cadena de condiciones (como un case de booleanos) y siempre termina con true categoría default.

edad = 16

# if/unless básicos
if edad >= 18 do
  IO.puts("Mayor")
else
  IO.puts("Menor")
end

unless edad >= 18 do
  IO.puts("Todavía menor")
end

# Forma abreviada (do: ...)
etiqueta = if edad >= 18, do: "adulto", else: "menor"
IO.puts(etiqueta)

# cond: cadena de condiciones
defmodule Clasifica do
  def nota(n) do
    cond do
      n >= 18 -> "A"
      n >= 14 -> "B"
      n >= 10 -> "C"
      true -> "D"
    end
  end
end

IO.puts(Clasifica.nota(16))     # B

# IMPORTANTE: if/unless/cond son funcionales: devuelven valor

En Elixir, únicamente false y nil son falsos: 0, "" y [] son VERDADEROS. No escribas if lista para saber si está vacía; usa case o Enum.empty?.

Funciones con múltiples cláusulas

La característica más elixirseca: puedes definir varias cabeceras de la misma función con patrones distintos; Elixir elige la primera que coincida. Esto reemplaza a if/case y a la herencia por completo.

defmodule Forma do
  # Misma función, varias cláusulas, patrones en los argumentos
  def area_circulo({:circulo, radio}), do: 3.14159 * radio * radio
  def rectangulo({:rectangulo, base, altura}), do: base * altura
  def desconocida(_), do: {:error, :no_soporta}

  # Cláusulas numéricas (pattern matching literal)
  def fizzbuzz(n) when rem(n, 15) == 0, do: "FizzBuzz"
  def fizzbuzz(n) when rem(n, 3) == 0, do: "Fizz"
  def fizzbuzz(n) when rem(n, 5) == 0, do: "Buzz"
  def fizzbuzz(n), do: n

  # Cláusula de lista: base y recursión
  def suma([]), do: 0
  def suma([h | t]), do: h + suma(t)
end

IO.inspect(Forma.area_circulo({:circulo, 2.0}))    # 12.56636
IO.inspect(Forma.rectangulo({:rectangulo, 3, 4}))  # 12
IO.inspect(Forma.rectangulo({:otra, 1}))           # {:error, :no_soporta}
IO.inspect(Forma.fizzbuzz(15))                     # "FizzBuzz"
IO.inspect(Forma.suma([1, 2, 3, 4]))               # 10

Ordenar las cláusulas de la más específica a la más general. Este estilo elimina los if/else a favor de patrones descriptivos y es difícil de leer en otros lenguajes: es el corazón de Elixir.

Guards: when

Los guards (when) añaden condiciones a una cláusula con un conjunto permitido de predicados. Se usan en case, def, fn y cond para acotar cuándo encaja un patrón.

defmodule Guardado do
  def tipo(n) when is_integer(n) and n > 0, do: "entero positivo"
  def tipo(n) when is_integer(n), do: "entero no positivo"
  def tipo(s) when is_binary(s) and String.length(s) > 5, do: "string largo"
  def tipo(_), do: "otro"

  # Guard en case
  def describe_resultado(x) do
    case x do
      {:ok, v} when v > 100 -> "grande éxito: #{v}"
      {:ok, v} -> "éxito: #{v}"
      {:error, _} -> "falló"
      _ -> "desconocido"
    end
  end

  # Guard con longitud
  def corto_o_largo(cadena) when byte_size(cadena) < 5, do: :corto
  def corto_o_largo(_), do: :largo
end

IO.inspect(Guardado.tipo(5))                     # "entero positivo"
IO.inspect(Guardado.tipo(-1))                    # "entero no positivo"
IO.inspect(Guardado.tipo("abcdefg"))             # "string largo"
IO.inspect(Guardado.describe_resultado({:ok, 150}))

Recurrencia (recursión) en práctica

En Elixir no hay bucles con variables mutables: la iteración es recursión (optimizada con tail call) o, mejor, funciones de Enum. Veamos la recursión directa y sus patrones.

defmodule Recursivo do
  # Suma
  def suma([]), do: 0
  def suma([h | t]), do: h + suma(t)

  # Longitud
  def longitud([]), do: 0
  def longitud([_h | t]), do: 1 + longitud(t)

  # Tail recursion con acumulador (no crece la pila)
  def suma_tail(lista), do: aux(lista, 0)
  defp aux([], acc), do: acc
  defp aux([h | t], acc), do: aux(t, acc + h)

  # Números de Fibonacci
  def fib(0), do: 0
  def fib(1), do: 1
  def fib(n), do: fib(n - 1) + fib(n - 2)

  # Invertir lista
  def inversa(lista), do: inv(lista, [])
  defp inv([], acc), do: acc
  defp inv([h | t], acc), do: inv(t, [h | acc])
end

IO.inspect(Recursivo.suma([1, 2, 3]))          # 6
IO.inspect(Recursivo.suma_tail([10, 20, 30]))  # 60
IO.inspect(Recursivo.longitud([1, 2, 3, 4]))   # 4
IO.inspect(Recursivo.fib(10))                  # 55
IO.inspect(Recursivo.inversa([1, 2, 3]))       # [3, 2, 1]

La "tail call" hace que la recursión con acumulador sea tan eficiente como un bucle. Aun así, para el 90% de los casos usa Enum.map/Enum.reduce. Aprende ambas: la recursión te da las bases de cómo funciona.

Ejercicios prácticos

Practica en orden: cada ejercicio asume lo aprendido en el anterior.

  1. case de estados: escribe una función que con case describa :conectado, :desconectado y otros estados.
  2. cond: clasifica un número en negativo/cero/positivo/uno con cond.
  3. Múltiples cláusulas: implementa saludar/1 para :mañana, :noche y cualquier _.
  4. Guardas: crea clasifica(n)` que con guards devuelva "grande", "mediano" o "pequeño" según rangos.
  5. Tail recursion: implementa cuenta_impares(lista) acumulando con tail call.
