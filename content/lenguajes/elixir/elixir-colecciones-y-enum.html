Listas

Las listas en Elixir son enlazadas e inmutables: agregar al frente es barato ([nueva | lista]); recorrerlas es la operación natural. El pattern matching extrae cabeza y cola.

lista = [1, 2, 3]

# Construcción: cons y concatenación
con = 0 :: [1, 2]        # [0, 1, 2]
concat = [1, 2] ++ [3]   # [1, 2, 3]
IO.inspect(con)
IO.inspect(concat)

# Acceso por índice y head/tail
IO.inspect(hd([9, 8, 7]))        # 9
IO.inspect(tl([9, 8, 7]))        # [8, 7]
IO.inspect(List.first([5, 6]))   # 5

# Patrón cabeza|cola
defmodule Demo do
  def primera([h | _]), do: h
end
IO.inspect(Demo.primera([10, 20]))   # 10

# Otras utilidades
IO.inspect(Enum.take([1,2,3,4], 2))   # [1, 2]
IO.inspect(List.zip([[1,2],[10,20]])) # [{1,10},{2,20}]
IO.inspect(List.delete([1,2,3], 2))   # [1, 3]

La cabeza es O(1); el final (++) es O(n), costoso en bucles. Si necesitas agregar al final muchas veces, usa Enum.reverse o acumula al frente y revierte al terminar.

Maps, keywords y conjuntos

Los Map son diccionarios con claves cualquier tipo; los Keyword son listas de pares átomo-valor (usadas para opciones); y MapSet es el conjunto sin duplicados.

# Map
usuario = %{nombre: "Ana", edad: 30, ciudad: "Lima"}
IO.inspect(Map.get(usuario, :edad))           # 30
IO.inspect(Map.get(usuario, :pais, "?)"))     # "no existe"
IO.inspect(Map.keys(usuario))                 # [:nombre, :edad, :ciudad]
IO.inspect(Map.has_key?(usuario, :nombre))    # true

# Actualizar / agregar (nuevo map, inmutable)
nuevo = Map.put(usuario, :edad, 31)
IO.inspect(usuario.edad)                      # 30 (no cambió)
IO.inspect(nuevo.edad)                        # 31

# Keyword list (ops de funciones, orden importa)
ops = [ancho: 100, color: "azul", ancho: 200]
IO.inspect(Keyword.fetch!(ops, :color))       # "azul"
IO.inspect(Keyword.get(ops, :ancho))          # 100 (primero)
IO.inspect(Keyword.get_values(ops, :ancho))   # [100, 200] todos

# MapSet
conj = MapSet.new([1, 2, 2, 3])
IO.inspect(MapSet.to_list(conj))              # [1, 2, 3]
IO.inspect(MapSet.member?(conj, 2))           # true

# Enum opera sobre todos: map, lista, keyword, mapset

¿Map o Keyword? Si la opción aparece una vez y el orden importa (caso típico: config de funciones) usa Keyword; para datos asociativos opacos usa Map (claves no solo átomos, y acceso O(log n)).

El módulo Enum

Enum es la caja de herramientas para todo iterable: map, filter, reduce, sort, group_by, count... Combinado con pipes es la forma natural de transformar datos.

nums = [1, 2, 3, 4, 5]

# transformar
IO.inspect(Enum.map(nums, &(&1 * 10)))        # [10,20,30,40,50]
IO.inspect(Enum.filter(nums, &(&1 > 2)))      # [3,4,5]
IO.inspect(Enum.reject(nums, &rem(&1, 2) == 0))  # [1,3,5]

# reducir
IO.inspect(Enum.reduce(nums, 0, &(&1 + &2)))  # 15
IO.inspect(Enum.sum(nums))                     # 15
IO.inspect(Enum.product([1,2,3]))              # 6
IO.inspect(Enum.count(nums))                   # 5

# seleccionar
IO.inspect(Enum.find(nums, &(&1 > 3)))        # 4 (primero que cumple)
IO.inspect(Enum.max(nums))                     # 5
IO.inspect(Enum.min(nums))                     # 1

# agrupar
IO.inspect(Enum.group_by([1,2,3,4,5], &rem(&1, 2)))
# %{0 => [2,4], 1 => [1,3,5]}

# ordenar y tomar
IO.inspect(Enum.sort([3,1,2], :desc))          # [3,2,1]
IO.inspect(Enum.take([1..5], 3))               # toma 3 primeros

# trampa: with_enumerable? y enumerables perezosos (Stream)
enum_pipe =
  [1, 2, 3, 4]
  |> Enum.filter(&(&1 > 1))
  |> Enum.map(&(&1 * 100))
IO.inspect(enum_pipe)                          # [200,300,400]

Memoriza estos cinco: map, filter, reduce, find, sort. Cubren casi todo análisis de datos. Para datos INFINITOS o muy grandes, la versión perezosa es Stream (misma API).

Comprehensions: for

El for de Elixir es una comprensión: genera listas combinando generadores, filtros (when) y una expresión de salida. Es el otro gran pilar para construir colecciones.

# Comprensión básica
cuadrados = for n <- [1, 2, 3, 4], do: n * n
IO.inspect(cuadrados)               # [1, 4, 9, 16]

# Con filtro when
pares = for n <- 1..20, rem(n, 2) == 0, do: n
IO.inspect(pares)                   # [2,4,...,20]

# Múltiples generadores (producto cartesiano)
pares_cart = for x <- 1..3, y <- 1..2, do: {x, y}
IO.inspect(pares_cart)

# Con opcion into: generar otros tipos
mapa = for {k, v} <- [a: 1, b: 2], into: %{}, do: {k, v * 10}
IO.inspect(mapa)                    # %{a: 10, b: 20}

# String generado con into
texto = for c <- "abc", into: "", do: String.upcase(<< c >>)
IO.inspect(texto)                   # "ABC"

# Filtrar por dos condiciones
nums_validos =
  for x <- 1..10,
      rem(x, 2) == 0,
      x < 8,
      do: x * x
IO.inspect(nums_validos)            # [4, 16, 36]

Los generadores usan <- (flecha hacia el contenedor), y la opción into: permite recolectar en Maps, Strings, Listas... En comparaciones, "for" en Elixir SIEMPRE devuelve una colección.

Structs y su integración con Enum

Las Structs son maps con esquema fijo (campos por defecto + nombre de módulo). Se definen con defstruct, y el pattern matching + Enum funcionan igual.

defmodule Producto do
  defstruct nombre: "sin nombre", precio: 0.0

  def con_impuesto(p, tasa), do: %{p | precio: p.precio * (1 + tasa)}
end

# Crear
lapiz = %Producto{nombre: "Lápiz", precio: 1.5}
IO.inspect(lapiz.nombre)                 # "Lápiz"
IO.inspect(lapiz)                        # %Producto{nombre: ..., precio: 1.5}

# Update
lapiz_caro = Producto.con_impuesto(lapiz, 0.18)
IO.inspect(lapiz_caro.precio)            # 1.77

# Pattern matching sobre struct
case lapiz do
  %Producto{precio: p} when p > 1 -> IO.puts("caro")
  %Producto{} -> IO.puts("barato")
end

# Enumerables también sirven con structs (es copy de map)
precios = [%Producto{nombre: "e", precio: 3}, %Producto{precio: 5}]
total = Enum.map(precios, & &1.precio) |> Enum.sum
IO.inspect(total)                        # 8.0

# En Structs no puedes poner claves que no existan: estricto
# %Producto{extra: 1}  →  KeyError en tiempo de compilación

Los Structs son a los Maps lo que las "clases sin métodos" a los diccionarios: esquema garantizado. Encima el campo __struct__ del map permite polimorfismo vía protocols cuando avances.
